export const title = "The Glass Astrarium — Passacaglia in Three Verses";

/**
 * A 24-beat passacaglia over an eight-note ground (D minor with a Phrygian
 * flat-two and a diminished leading-tone turn). Three verses: emergence,
 * weaving, bloom; then a caesura, then a coda that returns to the first note.
 *
 * The picture is an astrarium: one full revolution of the clock hand is the
 * whole piece. Three counter-rotating orbits carry the three upper voices —
 * time drives their angle, pitch drives their radius — so their drawn trails
 * ARE the counterpoint. An eight-toothed gear at the centre is the ground.
 * Ear and eye read the same clock (audioContext.currentTime), and every note
 * onset is recorded once and used by both the scheduler and the renderer.
 */
export function start({ audioContext, canvas, duration, random }) {
  const ac = audioContext;
  const ctx = canvas.getContext("2d");
  const W = canvas.width || 1080;
  const H = canvas.height || 1350;
  const TAU = Math.PI * 2;
  const CX = W / 2;
  const CY = H * 0.445;

  const clamp = (x, a, b) => (x < a ? a : x > b ? b : x);
  const smooth = (a, b, x) => {
    const u = clamp((x - a) / (b - a), 0, 1);
    return u * u * (3 - 2 * u);
  };
  const bump = (x, a, b, c) => smooth(a, b, x) * (1 - smooth(b, c, x));
  const hsla = (h, s, l, a) =>
    `hsla(${((h % 360) + 360) % 360},${s}%,${l}%,${clamp(a == null ? 1 : a, 0, 1)})`;
  const m2f = (m) => 440 * Math.pow(2, (m - 69) / 12);
  const polar = (r, a) => [Math.cos(a) * r, Math.sin(a) * r];

  if (ac.state === "suspended" && ac.resume) ac.resume();
  const D = duration > 0 ? duration : 15;
  const B = D / 24;                        // one beat = 1/24 of the piece
  const t0 = ac.currentTime + 0.06;
  const at = (b) => t0 + b * B;

  // ---------------------------------------------------------------- textures
  const NT = 4096;
  const noise = new Float32Array(NT);
  for (let i = 0; i < NT; i++) noise[i] = random() * 2 - 1;
  const nz = (i) => noise[i & (NT - 1)];

  // ---------------------------------------------------------------- timbres
  const mkWave = (a) =>
    ac.createPeriodicWave(new Float32Array(a.length), Float32Array.from(a));
  const W_GLASS = mkWave([0, 1, 0.30, 0.13, 0.055, 0.021, 0.008]);
  const W_REED = mkWave([0, 1, 0.16, 0.55, 0.10, 0.30, 0.06, 0.15, 0.03]);
  const W_BASS = mkWave([0, 1, 0.34, 0.10, 0.04]);

  // ---------------------------------------------------------------- audio graph
  const bus = ac.createGain();
  bus.gain.value = 1;
  const comp = ac.createDynamicsCompressor();
  comp.threshold.value = -18;
  comp.knee.value = 20;
  comp.ratio.value = 4;
  comp.attack.value = 0.004;
  comp.release.value = 0.25;
  const master = ac.createGain();
  bus.connect(comp);
  comp.connect(master);

  const conv = ac.createConvolver();
  conv.buffer = impulse(2.9, 2.4);
  conv.normalize = true;
  const wet = ac.createGain(); wet.gain.value = 0.30;
  const dry = ac.createGain(); dry.gain.value = 0.80;
  master.connect(dry); dry.connect(ac.destination);
  master.connect(conv); conv.connect(wet); wet.connect(ac.destination);

  // dotted-eighth echo with a darkening feedback loop
  const delay = ac.createDelay(1.0);
  delay.delayTime.value = B * 0.75;
  const dlp = ac.createBiquadFilter(); dlp.type = "lowpass"; dlp.frequency.value = 2600;
  const dfb = ac.createGain(); dfb.gain.value = 0.30;
  const dw = ac.createGain(); dw.gain.value = 0.26;
  delay.connect(dlp); dlp.connect(dfb); dfb.connect(delay);
  delay.connect(dw); dw.connect(bus);

  // a single, very quiet shared vibrato for the sustained voices
  const lfo = ac.createOscillator(); lfo.frequency.value = 5.15;
  const vib = ac.createGain(); vib.gain.value = 3.2;
  lfo.connect(vib);
  const vibSink = ac.createGain(); vibSink.gain.value = 0;
  vib.connect(vibSink); vibSink.connect(bus);
  lfo.start(t0); lfo.stop(t0 + D + 1);

  // ---------------------------------------------------------------- dynamics
  const mg = master.gain;
  mg.setValueAtTime(0.0001, t0);
  mg.linearRampToValueAtTime(0.86, at(0.55));
  mg.setValueAtTime(0.86, at(17.0));
  mg.linearRampToValueAtTime(1.00, at(20.2));
  mg.setValueAtTime(1.00, at(20.85));
  mg.linearRampToValueAtTime(0.20, at(21.03));   // the caesura
  mg.setValueAtTime(0.20, at(21.82));
  mg.linearRampToValueAtTime(0.80, at(22.25));   // coda bloom
  mg.setValueAtTime(0.80, at(22.80));
  mg.linearRampToValueAtTime(0.0001, at(23.95));

  // ------------------------------------------------- one memory, ear and eye
  const notes = [[], [], [], [], []];   // 0 bass · 1 tenor · 2 altus · 3 cantus · 4 celesta
  const events = [];
  function register(v, when, durB, m, amp) {
    const t = when - t0;
    notes[v].push({ t, m });
    events.push({ t, v, m, a: amp, s: random() });
  }
  function pitchAt(v, t) {
    const L = notes[v];
    if (!L.length) return 0;
    let i = 0;
    while (i + 1 < L.length && L[i + 1].t <= t) i++;
    const cur = L[i], nxt = L[i + 1];
    let m = cur.m;
    if (nxt) {
      const g = 0.26 * B;                  // a gentle portamento anticipation
      if (t > nxt.t - g) {
        const u = clamp((t - (nxt.t - g)) / g, 0, 1);
        m = cur.m + (nxt.m - cur.m) * u * u * (3 - 2 * u);
      }
    }
    return m;
  }

  let Rcur = 1;
  const VR = [150, 250, 345];                         // orbital radii
  const SPD = [TAU / 8, -TAU / 16, TAU / 24];        // 3 : -1.5 : 1 revolutions
  const PHASE = [0.45, 2.15, 4.25];
  const MID = [58, 68, 80];
  function pos(vi, tt) {
    const a = -Math.PI / 2 + (tt / B) * SPD[vi] + PHASE[vi];
    const m = pitchAt(vi + 1, tt);
    const r = (VR[vi] + (m - MID[vi]) * 2.4) * Rcur;
    return [Math.cos(a) * r, Math.sin(a) * r];
  }

  // ---------------------------------------------------------------- helpers
  function impulse(secs, decay) {
    const sr = ac.sampleRate;
    const len = Math.max(1, Math.floor(sr * secs));
    const buf = ac.createBuffer(2, len, sr);
    for (let ch = 0; ch < 2; ch++) {
      const d = buf.getChannelData(ch);
      let lp = 0, lp2 = 0;
      for (let i = 0; i < len; i++) {
        const n = nz(i + ch * 1337 + (i >> 3));
        lp += 0.25 * (n - lp);
        lp2 += 0.055 * (lp - lp2);
        d[i] = (lp2 * 1.7 + n * 0.13) * Math.pow(1 - i / len, decay);
      }
      const taps = [0.009, 0.017, 0.029, 0.043, 0.061, 0.083, 0.107, 0.137];
      for (let k = 0; k < taps.length; k++) {
        const i = Math.floor(sr * taps[k] * (1 + ch * 0.09));
        if (i < len) d[i] += (0.40 - k * 0.043) * (k % 2 ? -1 : 1);
      }
      const f = Math.floor(sr * 0.005);
      for (let i = 0; i < f && i < len; i++) d[i] *= i / f;
    }
    return buf;
  }

  function adsr(g, when, dur, peak, a, d, sus, r) {
    const A = Math.min(a, dur * 0.30);
    const Dd = Math.min(d, dur * 0.30);
    const R = Math.min(r, dur * 0.50);
    const hold = Math.max(A + Dd + 0.01, dur - R);
    const p = g.gain;
    p.setValueAtTime(0.0001, when);
    p.linearRampToValueAtTime(peak, when + A);
    p.linearRampToValueAtTime(peak * sus, when + A + Dd);
    p.setValueAtTime(peak * sus, when + hold);
    p.exponentialRampToValueAtTime(0.0001, when + dur);
  }

  function send(node, p) {
    if (p && ac.createStereoPanner) {
      const sp = ac.createStereoPanner();
      sp.pan.value = clamp(p, -1, 1);
      node.connect(sp); sp.connect(bus);
    } else node.connect(bus);
  }

  const COLL = [57, 58, 60, 61, 62, 63, 64, 65, 67, 69, 70, 72, 73, 74,
                76, 77, 79, 81, 82, 84, 85, 86, 88];
  function step(from, to) {
    const i = COLL.indexOf(from), j = COLL.indexOf(to);
    if (i < 0 || j < 0) return from;
    if (j > i) return COLL[Math.min(i + 1, j)];
    if (j < i) return COLL[Math.max(i - 1, j)];
    return COLL[clamp(i + (random() < 0.5 ? 1 : -1), 0, COLL.length - 1)];
  }

  // ---------------------------------------------------------------- voices
  function glass(when, durB, m, amp, p) {           // cantus: bowed glass
    const dur = durB * B, f = m2f(m);
    const lp = ac.createBiquadFilter(); lp.type = "lowpass"; lp.Q.value = 0.6;
    lp.frequency.setValueAtTime(5200, when);
    lp.frequency.exponentialRampToValueAtTime(1700, when + dur);
    const g = ac.createGain();
    adsr(g, when, dur, amp, 0.16 * B, 0.26 * B, 0.70, 0.75 * B);
    for (let k = 0; k < 2; k++) {
      const o = ac.createOscillator();
      o.setPeriodicWave(W_GLASS);
      o.frequency.value = f;
      o.detune.value = k ? 6.5 : -2;
      vib.connect(o.detune);
      o.connect(lp);
      o.start(when); o.stop(when + dur + 0.06);
    }
    lp.connect(g); send(g, p);
    register(3, when, durB, m, amp);
  }

  function reed(when, durB, m, amp, p) {            // altus: a hollow reed
    const dur = durB * B, f = m2f(m);
    const lp = ac.createBiquadFilter(); lp.type = "lowpass"; lp.Q.value = 0.9;
    lp.frequency.setValueAtTime(2400, when);
    lp.frequency.exponentialRampToValueAtTime(1050, when + dur);
    const g = ac.createGain();
    adsr(g, when, dur, amp, 0.06 * B, 0.22 * B, 0.62, 0.42 * B);
    const o = ac.createOscillator();
    o.setPeriodicWave(W_REED); o.frequency.value = f;
    vib.connect(o.detune);
    o.connect(lp); lp.connect(g); send(g, p);
    o.start(when); o.stop(when + dur + 0.05);
    register(2, when, durB, m, amp);
  }

  function bass(when, m, durB, coda) {              // the ground
    const dur = durB * B, f = m2f(m), peak = coda ? 0.16 : 0.12;
    const lp = ac.createBiquadFilter(); lp.type = "lowpass";
    lp.frequency.value = coda ? 520 : 250; lp.Q.value = 0.7;
    const g = ac.createGain();
    adsr(g, when, dur, peak, 0.12 * B, 0.30 * B, 0.78, 1.0 * B);
    const o = ac.createOscillator(); o.setPeriodicWave(W_BASS); o.frequency.value = f;
    const sub = ac.createOscillator(); sub.type = "sine"; sub.frequency.value = f / 2;
    const sg = ac.createGain(); sg.gain.value = 0.55;
    o.connect(lp); sub.connect(sg); sg.connect(lp); lp.connect(g); send(g, 0);
    o.start(when); sub.start(when);
    o.stop(when + dur + 0.05); sub.stop(when + dur + 0.05);
    register(0, when, durB, m, peak);
  }

  const RATIOS = [1, 2.004, 3.013, 4.21];          // inharmonic: real glass
  const PAMPS = [1, 0.40, 0.20, 0.09];
  function bell(when, m, amp, p, durB, fx, v) {     // tenor + celesta
    when = Math.max(t0, when + (random() - 0.5) * 0.016 * B);
    const f = m2f(m), dur = durB * B;
    const mix = ac.createGain(); mix.gain.value = 1;
    for (let k = 0; k < RATIOS.length; k++) {
      const o = ac.createOscillator(); o.type = "sine"; o.frequency.value = f * RATIOS[k];
      const g = ac.createGain();
      const end = dur * Math.max(0.28, 1 - k * 0.19);
      g.gain.setValueAtTime(0.0001, when);
      g.gain.linearRampToValueAtTime(amp * PAMPS[k], when + 0.004 * B);
      g.gain.exponentialRampToValueAtTime(0.0001, when + end);
      o.connect(g); g.connect(mix);
      o.start(when); o.stop(when + end + 0.03);
    }
    send(mix, p);
    if (fx > 0) { const s = ac.createGain(); s.gain.value = fx; mix.connect(s); s.connect(delay); }
    register(v, when, durB, m, amp);
  }

  function pip(when, amp) {                         // the escapement
    const o = ac.createOscillator(); o.type = "sine"; o.frequency.value = 2637;
    const g = ac.createGain();
    g.gain.setValueAtTime(0.0001, when);
    g.gain.linearRampToValueAtTime(amp, when + 0.003);
    g.gain.exponentialRampToValueAtTime(0.0001, when + 0.10);
    o.connect(g); send(g, 0);
    o.start(when); o.stop(when + 0.13);
  }

  // ---------------------------------------------------------------- THE SCORE
  // Eight-beat ground: D D F G A G E♭ C♯ → (back to D). Repeated, then broken
  // at the caesura so the last thing heard is the note the piece began on.
  const GROUND = [38, 38, 41, 43, 45, 43, 39, 37];
  const CHORD = [
    [50, 53, 57, 60],  // i    Dm
    [50, 53, 57, 60],  // i
    [53, 57, 60, 65],  // III  F
    [55, 58, 62, 67],  // iv   Gm
    [57, 61, 64, 69],  // V    A
    [55, 58, 62, 67],  // iv
    [51, 55, 58, 63],  // ♭II  E♭
    [49, 52, 55, 58],  // vii° C♯dim
  ];

  for (let b = 0; b <= 20; b++) bass(at(b), GROUND[b % 8], 0.98, false);
  bass(at(22), 38, 2.2, true);

  const CANTUS = [
    [0, 2, 74], [2, 2, 77], [4, 2, 81], [6, 1, 79], [7, 1, 76],
    [8, 2, 81], [10, 2, 84], [12, 1, 82], [13, 1, 81], [14, 1, 82], [15, 1, 79],
    [16, 1.5, 84], [17.5, 0.5, 86], [18, 1, 88], [19, 1, 86], [20, 1, 85],
    [22, 2.15, 74],                         // return
  ];
  for (const [b, d, m] of CANTUS) {
    const pan = clamp(-0.28 + 0.56 * (b / 24) + 0.08 * Math.sin(b * 0.7), -0.6, 0.6);
    const amp = b < 8 ? 0.085 : b < 16 ? 0.098 : b < 22 ? 0.112 : 0.090;
    glass(at(b), d, m, amp, pan);
  }

  // Verse I: three long guide tones only.
  reed(at(0), 2, 62, 0.050, -0.32);
  reed(at(4), 2, 69, 0.055, -0.32);
  reed(at(7), 1, 61, 0.046, -0.32);

  // Verses II–III: eighth-note counterpoint, sixteenths at the peak.
  const ALT = [62, 65, 65, 67, 69, 67, 63, 61,
               62, 65, 69, 70, 73, 70, 70, 67,
               69, 72, 74, 70, 73, 74];
  for (let b = 8; b <= 20; b++) {
    const t = ALT[b], nx = ALT[b + 1], mid = step(t, nx);
    const a = b < 16 ? 0.062 : 0.075;
    reed(at(b), 0.5, t, a, -0.34);
    if (b >= 18) reed(at(b + 0.25), 0.25, step(t, mid), a * 0.45, -0.34);
    reed(at(b + 0.5), 0.5, mid, a * 0.80, -0.34);
    if (b >= 18) reed(at(b + 0.75), 0.25, step(mid, nx), a * 0.45, -0.34);
  }
  reed(at(22), 2.15, 74, 0.052, -0.30);

  // Tenor: the clockwork. 2 sparse bells → triplets against duplets → sixteenths.
  const ORD2 = [0, 2], ORD3 = [0, 2, 1], ORD4 = [0, 2, 1, 3];
  for (let b = 0; b <= 20; b++) {
    let n, offs, ord, amp;
    if (b < 8) {
      if (b !== 0 && b !== 4 && b !== 7) continue;
      n = 2; offs = [0, 0.5]; ord = ORD2; amp = 0.052;
    } else if (b < 16) {
      n = 3; offs = [0, 1 / 3, 2 / 3]; ord = ORD3; amp = 0.070;
    } else {
      n = 4; offs = [0, 0.25, 0.5, 0.75]; ord = ORD4; amp = 0.082;
    }
    const set = CHORD[b % 8];
    for (let k = 0; k < n; k++) {
      let m = set[(ord[k] + b) % set.length];
      if (n === 4 && k === 3 && b % 2 === 0) m += 12;
      bell(at(b + offs[k]), m, amp * (0.85 + 0.30 * random()),
           k % 2 ? 0.48 : -0.48, 1.5, 0.22, 1);
    }
  }

  // Celesta shimmer, only in the bloom.
  for (let b = 16; b <= 20; b++) {
    const set = CHORD[b % 8].map((m) => m + 24);
    const count = b === 20 ? 5 : 3;
    for (let k = 0; k < count; k++) {
      const off = clamp(0.12 + k * (0.8 / count) + (random() - 0.5) * 0.07, 0, 0.95);
      const m = set[(random() * set.length) | 0] + (random() < 0.28 ? 12 : 0);
      bell(at(b + off), m, 0.030 + 0.022 * random(),
           (random() - 0.5) * 1.5, 1.9, 0.55, 4);
    }
  }

  // The hush: one suspended breath.
  bell(at(21.05), 86, 0.045, 0, 3.2, 0.70, 4);

  // Coda: a D-minor arpeggio and three falling embers back down to D5.
  [62, 65, 69, 74].forEach((m, i) =>
    bell(at(22 + i * 0.42), m, 0.055 - i * 0.007, i % 2 ? 0.40 : -0.40, 2.4, 0.40, 1));
  [[22.35, 86], [22.80, 81], [23.25, 74]].forEach(([b, m], i) =>
    bell(at(b), m, 0.038 - i * 0.008, (i - 1) * 0.4, 2.6, 0.60, 4));

  // Low gongs mark the three verses; a hair-thin pip marks every beat.
  [0, 8, 16].forEach((b, i) => bell(at(b), [38, 43, 45][i], 0.055, 0, 2.6, 0.18, 0));
  for (let b = 0; b <= 23; b++) pip(at(b), 0.0045 + 0.002 * (b / 24));

  // Air: a breathing band of filtered noise, plus a felt D drone.
  const airLen = Math.floor(ac.sampleRate * 3);
  const airBuf = ac.createBuffer(1, airLen, ac.sampleRate);
  const ad = airBuf.getChannelData(0);
  let alp = 0;
  for (let i = 0; i < airLen; i++) {
    const n = nz(i * 3 + 77);
    alp += 0.12 * (n - alp);
    ad[i] = alp * 3 + n * 0.15;
  }
  const air = ac.createBufferSource(); air.buffer = airBuf; air.loop = true;
  const abp = ac.createBiquadFilter(); abp.type = "bandpass"; abp.Q.value = 0.6;
  const ag = ac.createGain();
  air.connect(abp); abp.connect(ag); send(ag, 0);
  ag.gain.setValueAtTime(0.0001, t0);
  ag.gain.linearRampToValueAtTime(0.012, at(8));
  ag.gain.linearRampToValueAtTime(0.030, at(16));
  ag.gain.linearRampToValueAtTime(0.055, at(20.5));
  ag.gain.linearRampToValueAtTime(0.075, at(21.2));   // the held breath
  ag.gain.linearRampToValueAtTime(0.020, at(22.5));
  ag.gain.linearRampToValueAtTime(0.0001, at(24.1));
  abp.frequency.setValueAtTime(320, t0);
  abp.frequency.linearRampToValueAtTime(1400, at(20.5));
  abp.frequency.linearRampToValueAtTime(380, at(24));
  air.start(t0); air.stop(at(24.2));

  const drone = ac.createOscillator(); drone.type = "sine"; drone.frequency.value = m2f(38);
  const dg = ac.createGain();
  dg.gain.setValueAtTime(0.0001, t0);
  dg.gain.linearRampToValueAtTime(0.018, at(8));
  dg.gain.linearRampToValueAtTime(0.030, at(20));
  dg.gain.linearRampToValueAtTime(0.010, at(22));
  dg.gain.linearRampToValueAtTime(0.0001, at(24));
  drone.connect(dg); send(dg, 0);
  drone.start(t0); drone.stop(at(24.2));

  // ---------------------------------------------------------------- PICTURE
  const stars = [];
  for (let i = 0; i < 230; i++) stars.push({
    x: random() * W, y: random() * H,
    r: 0.35 + random() * 1.5, a: 0.05 + random() * 0.30,
    ph: random() * TAU, sp: 0.3 + random() * 0.9, d: 0.25 + random() * 0.7,
  });
  const motes = [];
  for (let i = 0; i < 64; i++) motes.push({
    x: random() * W, y: random() * H,
    s: 0.2 + random() * 0.8, r: 0.6 + random() * 1.7,
    ph: random() * TAU, a: 0.04 + random() * 0.12,
  });
  const grain = [];
  for (let i = 0; i < 420; i++) grain.push([(random() * W) | 0, (random() * H) | 0]);

  function hueAt(p) {                         // indigo → amber → rose → indigo
    if (p < 0.33) return 222 + 10 * Math.sin(p * 10);
    if (p < 0.66) return 222 + (36 - 222) * ((p - 0.33) / 0.33);
    if (p < 0.87) return 36 + (346 - 36) * ((p - 0.66) / 0.21);
    return 346 + (222 - 346) * ((p - 0.87) / 0.13);
  }

  function draw() {
    const t = ac.currentTime - t0;
    const p = clamp(t / D, 0, 1);
    const b = t / B;
    const hue = hueAt(p);
    const build = smooth(0.0, 0.78, p);
    const climax = bump(p, 0.66, 0.84, 0.93);
    const hush = bump(p, 0.862, 0.882, 0.906);
    const collapse = smooth(0.925, 1.0, p);
    const bright = (0.30 + 0.62 * build + 0.42 * climax) * (1 - 0.72 * hush);
    const fade = 1 - collapse;
    const A = bright * fade;
    const verse = p < 1 / 3 ? 0 : p < 2 / 3 ? 1 : 2;
    Rcur = 1 - 0.10 * collapse;

    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.globalCompositeOperation = "source-over";
    ctx.globalAlpha = 1;

    const bg = ctx.createLinearGradient(0, 0, 0, H);
    bg.addColorStop(0, "#04070e");
    bg.addColorStop(0.52, "#080c16");
    bg.addColorStop(1, "#03050a");
    ctx.fillStyle = bg; ctx.fillRect(0, 0, W, H);

    const glow = ctx.createRadialGradient(CX, CY, 0, CX, CY, 640);
    glow.addColorStop(0, hsla(hue, 62, 24, 0.30 * bright));
    glow.addColorStop(0.45, hsla(hue, 58, 13, 0.14 * bright));
    glow.addColorStop(1, hsla(hue, 50, 8, 0));
    ctx.fillStyle = glow; ctx.fillRect(0, 0, W, H);

    for (const s of stars) {
