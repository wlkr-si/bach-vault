/**
 * TOCCATA & CANON IN D MINOR: THE ARCHITECTURE OF INFINITY

 * An original 15-second polyphonic masterpiece for Bach Bench.
 * Combines authentic Baroque counterpoint (Toccata, 3-Voice Fugue, Chromatic Stretto,
 * and Picardy Third Apotheosis) with high-density procedural synthesis (Pipe Organ, 
 * Flemish Cembalo, 32' Contra-Bourdon, Cathedral Impulse Reverb) and real-time
 * 3D Gothic Cathedral sacred geometry visualization on a 1080 × 1350 canvas.

export const title = "Toccata & Canon in D minor: The Architecture of Infinity";

export function start({ audioContext, canvas, duration, random }) {
  // Ensure portrait canvas dimensions
  canvas.width = 1080;
  canvas.height = 1350;
  const ctx = canvas.getContext("2d");

  // ============================================================================
  // 1. MASTER AUDIO ARCHITECTURE & SYNTHESIS ENGINE
  // ============================================================================
  const startTime = audioContext.currentTime + 0.05;

  // Master Dynamics & Output Bus
  const masterCompressor = audioContext.createDynamicsCompressor();
  masterCompressor.threshold.setValueAtTime(-14, startTime);
  masterCompressor.knee.setValueAtTime(8, startTime);
  masterCompressor.ratio.setValueAtTime(4.5, startTime);
  masterCompressor.attack.setValueAtTime(0.003, startTime);
  masterCompressor.release.setValueAtTime(0.12, startTime);

  const masterGain = audioContext.createGain();
  masterGain.gain.setValueAtTime(0.88, startTime);
  // Gentle fade-out right at the 15.0s benchmark cut
  masterGain.gain.setValueAtTime(0.88, startTime + 14.3);
  masterGain.gain.exponentialRampToValueAtTime(0.001, startTime + 14.98);

  const analyser = audioContext.createAnalyser();
  analyser.fftSize = 512;
  analyser.smoothingTimeConstant = 0.8;

  masterGain.connect(masterCompressor);
  masterCompressor.connect(analyser);
  analyser.connect(audioContext.destination);

  // Cathedral Convolver Reverb (Synthetic Impulse Response: Leipzig Thomaskirche model)
  const irLength = Math.floor(audioContext.sampleRate * 2.8);
  const irBuffer = audioContext.createBuffer(2, irLength, audioContext.sampleRate);
  const irL = irBuffer.getChannelData(0);
  const irR = irBuffer.getChannelData(1);
  const earlyReflections = [
    { t: 0.021, g: 0.65, p: -0.35 },
    { t: 0.038, g: 0.55, p: 0.4 },
    { t: 0.054, g: 0.45, p: -0.25 },
    { t: 0.076, g: 0.38, p: 0.3 },
    { t: 0.098, g: 0.30, p: -0.5 },
    { t: 0.125, g: 0.25, p: 0.45 }
  ];

  for (let i = 0; i < irLength; i++) {
    const t = i / audioContext.sampleRate;
    const decay = Math.exp(-t * 2.1);
    irL[i] = (random() * 2 - 1) * decay * 0.45;
    irR[i] = (random() * 2 - 1) * decay * 0.45;
  }
  earlyReflections.forEach(tap => {
    const idx = Math.floor(tap.t * audioContext.sampleRate);
    if (idx < irLength) {
      irL[idx] += tap.g * (1 - tap.p) * 0.5;
      irR[idx] += tap.g * (1 + tap.p) * 0.5;
    }
  });

  const reverbConvolver = audioContext.createConvolver();
  reverbConvolver.buffer = irBuffer;
  const reverbWetGain = audioContext.createGain();
  reverbWetGain.gain.setValueAtTime(0.42, startTime);
  reverbConvolver.connect(reverbWetGain);
  reverbWetGain.connect(masterGain);

  const dryGain = audioContext.createGain();
  dryGain.gain.setValueAtTime(0.78, startTime);
  dryGain.connect(masterGain);

  // Noise Buffer for Organ Pipe Wind Transient ("Chiff")
  const chiffLength = Math.floor(audioContext.sampleRate * 0.035);
  const chiffBuffer = audioContext.createBuffer(1, chiffLength, audioContext.sampleRate);
  const chiffData = chiffBuffer.getChannelData(0);
  for (let i = 0; i < chiffLength; i++) {
    chiffData[i] = (random() * 2 - 1) * Math.exp(-i / (chiffLength * 0.3));
  }

  // Instrument Voice Synthesizers
  function midiToFreq(m) {
    return 440 * Math.pow(2, (m - 69) / 12);
  }

  function playOrganNote(pitch, time, dur, vel = 0.8) {
    const f0 = midiToFreq(pitch);
    const noteGain = audioContext.createGain();
    noteGain.gain.setValueAtTime(0.0001, time);
    noteGain.gain.exponentialRampToValueAtTime(vel * 0.32, time + 0.02);
    noteGain.gain.setValueAtTime(vel * 0.28, time + Math.max(0.03, dur - 0.08));
    noteGain.gain.exponentialRampToValueAtTime(0.0001, time + dur + 0.08);

    // Filter modeling pipe resonance
    const filter = audioContext.createBiquadFilter();
    filter.type = "lowpass";
    filter.frequency.setValueAtTime(Math.min(9500, f0 * 6.5), time);
    filter.Q.setValueAtTime(1.8, time);

    // Principal 8', Octave 4', Superoctave 2', Mixture 12th
    const osc1 = audioContext.createOscillator();
    osc1.type = "sawtooth";
    osc1.frequency.setValueAtTime(f0, time);

    const osc2 = audioContext.createOscillator();
    osc2.type = "triangle";
    osc2.frequency.setValueAtTime(f0 * 2, time);

    const osc3 = audioContext.createOscillator();
    osc3.type = "sine";
    osc3.frequency.setValueAtTime(f0 * 3, time);

    const osc2Gain = audioContext.createGain();
    osc2Gain.gain.value = 0.45;
    const osc3Gain = audioContext.createGain();
    osc3Gain.gain.value = 0.22;

    osc1.connect(filter);
    osc2.connect(osc2Gain);
    osc2Gain.connect(filter);
    osc3.connect(osc3Gain);
    osc3Gain.connect(filter);

    // Chiff wind burst
    const chiffNode = audioContext.createBufferSource();
    chiffNode.buffer = chiffBuffer;
    const chiffFilter = audioContext.createBiquadFilter();
    chiffFilter.type = "bandpass";
    chiffFilter.frequency.setValueAtTime(Math.min(6000, f0 * 3.2), time);
    chiffFilter.Q.setValueAtTime(3.0, time);
    const chiffGain = audioContext.createGain();
    chiffGain.gain.setValueAtTime(vel * 0.18, time);
    chiffGain.gain.exponentialRampToValueAtTime(0.0001, time + 0.035);
    chiffNode.connect(chiffFilter);
    chiffFilter.connect(chiffGain);
    chiffGain.connect(filter);

    filter.connect(noteGain);
    noteGain.connect(dryGain);
    noteGain.connect(reverbConvolver);

    osc1.start(time);
    osc2.start(time);
    osc3.start(time);
    chiffNode.start(time);

    const stopTime = time + dur + 0.1;
    osc1.stop(stopTime);
    osc2.stop(stopTime);
    osc3.stop(stopTime);
    chiffNode.stop(time + 0.04);
  }

    const bandpass = audioContext.createBiquadFilter();
    bandpass.type = "bandpass";
    bandpass.frequency.setValueAtTime(Math.min(8000, f0 * 2.4), time);
    bandpass.Q.setValueAtTime(2.2, time);

    const oscA = audioContext.createOscillator();
    oscA.type = "sawtooth";
    oscA.frequency.setValueAtTime(f0, time);

    const oscB = audioContext.createOscillator();
    oscB.type = "sawtooth";
    oscB.frequency.setValueAtTime(f0 * 1.002, time); // Subtle harpsichord double-string detune

    oscA.connect(bandpass);
    oscB.connect(bandpass);
    bandpass.connect(noteGain);

    noteGain.connect(reverbConvolver);

    oscA.start(time);
    oscB.start(time);
    const stopTime = time + Math.max(0.2, dur);
    oscA.stop(stopTime);
    oscB.stop(stopTime);

    subGain.gain.value = 0.65;
    const octGain = audioContext.createGain();
    octGain.gain.value = 0.3;

    oscSub.connect(subGain);
    subGain.connect(noteGain);
    oscFund.connect(noteGain);
    oscOct.connect(octGain);
    octGain.connect(noteGain);

    noteGain.connect(dryGain);
    noteGain.connect(reverbConvolver);

    oscSub.start(time);
    oscFund.start(time);
    oscOct.start(time);
    const stopTime = time + dur + 0.2;
    oscSub.stop(stopTime);
    oscFund.stop(stopTime);
    oscOct.stop(stopTime);

      x: cx + x * scale,

    const activePitchClasses = new Set(activeNotes.map(n => n.pitch % 12));

    // Camera push-in towards the Rose Window
    const camZ = -520 + progress * 420;
    const shake = bassEnergy * (elapsed > 12.6 ? 12 : 5);
    const shakeX = (random() * 2 - 1) * shake;
    const shakeY = (random() * 2 - 1) * shake;

    // Picardy state detection (D Major resolution)
    const isPicardy = elapsed >= 12.68;

    // --- A. BACKGROUND & SACRED CATHEDRAL VAULT ---
    ctx.save();
    ctx.translate(shakeX, shakeY);

    // Deep cathedral atmospheric gradient
    const bgGrad = ctx.createRadialGradient(cx, cy, 60, cx, cy, H * 0.85);
    if (isPicardy) {
      bgGrad.addColorStop(0, "#2c1d04");
      bgGrad.addColorStop(0.35, "#150b1c");
      bgGrad.addColorStop(1, "#040207");
    } else {
      bgGrad.addColorStop(0, "#0e0817");
      bgGrad.addColorStop(0.45, "#07040d");
      bgGrad.addColorStop(1, "#020105");
    }
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, W, H);

    // --- B. VOLUMETRIC CREPUSCULAR GOD RAYS ---
    const rayCount = 16;
    const rayAngleOffset = elapsed * 0.12;
    ctx.save();
    ctx.globalCompositeOperation = "lighter";
    for (let i = 0; i < rayCount; i++) {
      const theta = (i / rayCount) * Math.PI * 2 + rayAngleOffset;
      const rayAlpha = (0.04 + bassEnergy * 0.08 + (isPicardy ? 0.14 : 0)) * (0.6 + 0.4 * Math.sin(elapsed * 3 + i));
      const rayGrad = ctx.createRadialGradient(cx, cy, 20, cx, cy, H * 0.7);
      if (isPicardy) {
        rayGrad.addColorStop(0, `rgba(255, 235, 170, ${rayAlpha * 1.5})`);
        rayGrad.addColorStop(0.4, `rgba(255, 190, 70, ${rayAlpha * 0.6})`);
        rayGrad.addColorStop(1, "rgba(255, 170, 50, 0)");
      } else {
        rayGrad.addColorStop(0, `rgba(180, 190, 255, ${rayAlpha * 1.2})`);
        rayGrad.addColorStop(0.4, `rgba(140, 70, 240, ${rayAlpha * 0.5})`);
        rayGrad.addColorStop(1, "rgba(50, 10, 100, 0)");
      }
      ctx.fillStyle = rayGrad;
      ctx.beginPath();
      ctx.moveTo(cx, cy);
      const beamSpread = 0.18 + bassEnergy * 0.1;
      ctx.arc(cx, cy, H * 0.85, theta - beamSpread, theta + beamSpread);
      ctx.closePath();
      ctx.fill();
    }
    ctx.restore();

    // --- C. 3D GOTHIC POINTED ARCHES (NAVE VAULTING) ---
    const ARCH_COUNT = 9;
    const naveWidth = 560;
    const naveHeight = 650;
    for (let i = ARCH_COUNT - 1; i >= 0; i--) {
      const archZ = (i * 210 - (elapsed * 90) % 210) + 120 - camZ;
      if (archZ <= 10) continue;

      const pLeft = project(-naveWidth, naveHeight, archZ);
      const pRight = project(naveWidth, naveHeight, archZ);
      const pApex = project(0, -naveHeight * 0.88, archZ);
      const pSpringL = project(-naveWidth, -naveHeight * 0.15, archZ);
      const pSpringR = project(naveWidth, -naveHeight * 0.15, archZ);

      const archAlpha = Math.max(0, Math.min(1, 1.4 - (archZ / 1700))) * (0.35 + bassEnergy * 0.45);
      const strokeColor = isPicardy ? `rgba(255, 220, 130, ${archAlpha})` : `rgba(140, 160, 255, ${archAlpha})`;

      ctx.lineWidth = Math.max(1, pLeft.scale * 3.5);
      ctx.strokeStyle = strokeColor;

      // Draw Ogival Pointed Gothic Arch
      ctx.beginPath();
      ctx.moveTo(pLeft.x, pLeft.y);
      ctx.lineTo(pSpringL.x, pSpringL.y);
      // Left gothic arc up to apex
      ctx.quadraticCurveTo(pSpringL.x + (pApex.x - pSpringL.x) * 0.25, pApex.y + 40, pApex.x, pApex.y);
      // Right gothic arc down from apex
      ctx.quadraticCurveTo(pSpringR.x - (pSpringR.x - pApex.x) * 0.25, pApex.y + 40, pSpringR.x, pSpringR.y);
      ctx.lineTo(pRight.x, pRight.y);
      ctx.stroke();

      // Tierceron rib vault ceiling diagonals
      if (i < ARCH_COUNT - 1) {
        const nextZ = archZ + 210;
        const pNextApex = project(0, -naveHeight * 0.88, nextZ);
        ctx.strokeStyle = isPicardy ? `rgba(255, 200, 100, ${archAlpha * 0.35})` : `rgba(160, 120, 240, ${archAlpha * 0.3})`;
        ctx.lineWidth = Math.max(0.7, pLeft.scale * 1.5);
        ctx.beginPath();
        ctx.moveTo(pApex.x, pApex.y);
        ctx.lineTo(pNextApex.x, pNextApex.y);
        ctx.moveTo(pSpringL.x, pSpringL.y);
        ctx.lineTo(pNextApex.x, pNextApex.y);
        ctx.moveTo(pSpringR.x, pSpringR.y);
        ctx.lineTo(pNextApex.x, pNextApex.y);
        ctx.stroke();
      }
    }

    // --- D. THE GREAT ROSE WINDOW (SACRED GEOMETRIC MANDALA) ---
    const roseZ = 1200 - camZ;
    const roseProj = project(0, -60, roseZ);
    const roseRadius = 260 * roseProj.scale;

    ctx.save();
    ctx.translate(roseProj.x, roseProj.y);

    // Glowing halo behind Rose Window
    const haloGrad = ctx.createRadialGradient(0, 0, roseRadius * 0.1, 0, 0, roseRadius * 1.35);
    if (isPicardy) {
      haloGrad.addColorStop(0, "rgba(255, 245, 180, 0.95)");
      haloGrad.addColorStop(0.3, "rgba(255, 180, 50, 0.45)");
      haloGrad.addColorStop(1, "rgba(255, 140, 20, 0)");
    } else {
      haloGrad.addColorStop(0, "rgba(180, 160, 255, 0.7)");
      haloGrad.addColorStop(0.35, "rgba(100, 50, 200, 0.3)");
      haloGrad.addColorStop(1, "rgba(40, 10, 80, 0)");
    }
    ctx.fillStyle = haloGrad;
    ctx.beginPath();
    ctx.arc(0, 0, roseRadius * 1.35, 0, Math.PI * 2);
    ctx.fill();

    // Outer Tracery Rings
    ctx.strokeStyle = isPicardy ? "#ffd54f" : "#9c88ff";
    ctx.lineWidth = Math.max(1.5, 3 * roseProj.scale);
    ctx.beginPath();
    ctx.arc(0, 0, roseRadius, 0, Math.PI * 2);
    ctx.stroke();

    ctx.lineWidth = Math.max(1, 1.8 * roseProj.scale);
    ctx.beginPath();
    ctx.arc(0, 0, roseRadius * 0.78, 0, Math.PI * 2);
    ctx.stroke();

    // 12 Outer Petal Medallions (Circle of Fifths Nodes)
    const nodeCoords = [];
    for (let i = 0; i < 12; i++) {
      const angle = (i / 12) * Math.PI * 2 - Math.PI / 2;
      const nx = Math.cos(angle) * (roseRadius * 0.88);
      const ny = Math.sin(angle) * (roseRadius * 0.88);
      const pitchClass = circleOfFifthsPitches[i];
      const isActive = activePitchClasses.has(pitchClass);
      nodeCoords.push({ x: nx, y: ny, angle, pitchClass, isActive });

      // Medallion Foil
      ctx.beginPath();
      ctx.arc(nx, ny, roseRadius * 0.088, 0, Math.PI * 2);
      if (isActive) {
        ctx.fillStyle = isPicardy ? "#ffffff" : "#ffeb3b";
        ctx.shadowColor = isPicardy ? "#ffc107" : "#00e5ff";
        ctx.shadowBlur = 18;
      } else {
        ctx.fillStyle = "rgba(25, 15, 45, 0.85)";
        ctx.shadowBlur = 0;
      }
      ctx.fill();
      ctx.stroke();

      // Pitch Class Notation Glyph inside active medallions
      if (isActive || roseProj.scale > 0.8) {
        ctx.fillStyle = isActive ? "#000000" : "rgba(200, 200, 255, 0.6)";
        ctx.font = `600 ${Math.max(9, Math.floor(13 * roseProj.scale))}px serif`;
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillText(pitchNames[pitchClass], nx, ny);
      }
    }
    ctx.shadowBlur = 0;

    // Harmonic Chord Connection Lines between active notes
    ctx.lineWidth = Math.max(1.2, 2.5 * roseProj.scale);
    for (let i = 0; i < nodeCoords.length; i++) {
      if (!nodeCoords[i].isActive) continue;
      for (let j = i + 1; j < nodeCoords.length; j++) {
        if (!nodeCoords[j].isActive) continue;
        ctx.strokeStyle = isPicardy ? "rgba(255, 235, 140, 0.85)" : "rgba(0, 229, 255, 0.75)";
        ctx.beginPath();
        ctx.moveTo(nodeCoords[i].x, nodeCoords[i].y);
        ctx.quadraticCurveTo(0, 0, nodeCoords[j].x, nodeCoords[j].y);
        ctx.stroke();
      }
    }

    // Inner Gothic Rosette (Six-Fold and Twelve-Fold Counter-Rotating Filigree)
    const innerSpin = elapsed * 0.25;
    ctx.lineWidth = Math.max(1, 1.6 * roseProj.scale);
    ctx.strokeStyle = isPicardy ? "rgba(255, 215, 80, 0.8)" : "rgba(160, 130, 255, 0.65)";
    for (let f = 0; f < 6; f++) {
      const a = (f / 6) * Math.PI * 2 + innerSpin;
      const fx = Math.cos(a) * (roseRadius * 0.42);
      const fy = Math.sin(a) * (roseRadius * 0.42);
      ctx.beginPath();
      ctx.arc(fx, fy, roseRadius * 0.24, 0, Math.PI * 2);
      ctx.stroke();
    }
    for (let f = 0; f < 12; f++) {
      const a = (f / 12) * Math.PI * 2 - innerSpin * 1.5;
      const fx = Math.cos(a) * (roseRadius * 0.22);
      const fy = Math.sin(a) * (roseRadius * 0.22);
      ctx.beginPath();
      ctx.arc(fx, fy, roseRadius * 0.1, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();

    // --- E. 3D POLYPHONIC COUNTERPOINT STREAMERS (VOICE RIBBONS) ---
    // Record current pitch per voice for kinetic 3D streamers
    for (let v = 0; v < 4; v++) {
      const vNotes = activeNotes.filter(n => n.v === v);
      let targetY = 120;
      let targetX = (v - 1.5) * 180;
      if (vNotes.length > 0) {
        const avgPitch = vNotes.reduce((acc, cur) => acc + cur.pitch, 0) / vNotes.length;
        targetY = -((avgPitch - 60) * 14) + (v === 3 ? 240 : 0);
      }
      voiceTrails[v].unshift({
        x: targetX,
        y: targetY,
        time: elapsed,
        hasNote: vNotes.length > 0
      });
      if (voiceTrails[v].length > TRAIL_LENGTH) voiceTrails[v].pop();

      // Render ribbon connecting points from rose window out into the nave
      if (voiceTrails[v].length > 2) {
        ctx.save();
        ctx.globalCompositeOperation = "lighter";
        ctx.beginPath();
        let started = false;
        for (let t = 0; t < voiceTrails[v].length; t++) {
          const pt = voiceTrails[v][t];
          const ptZ = (t * 42) + 150 - camZ;
          if (ptZ < 10) continue;
          const pr = project(pt.x, pt.y, ptZ);
          if (!started) {
            ctx.moveTo(pr.x, pr.y);
            started = true;
          } else {
            ctx.lineTo(pr.x, pr.y);
          }
        }
        const vCol = voiceColors[v];
        const alpha = (0.35 + bassEnergy * 0.4) * (vNotes.length > 0 ? 1.0 : 0.4);
        ctx.strokeStyle = `rgba(${vCol.rgb[0]}, ${vCol.rgb[1]}, ${vCol.rgb[2]}, ${alpha})`;
        ctx.lineWidth = Math.max(1.8, 5.5 * (1 - v / 4));
        ctx.stroke();
        ctx.restore();
      }
    }

    // --- F. FLOATING INCENSE MOTES & SOUND PARTICLES ---
    ctx.save();
    ctx.globalCompositeOperation = "lighter";
    particles.forEach(p => {
      p.z -= p.vz * (1 + bassEnergy * 1.5);
      if (p.z < 20) p.z += 1600;
      const pr = project(p.x, p.y + Math.sin(elapsed * 2 + p.phase) * 35, p.z - camZ);
      if (pr.scale > 0) {
        const pSize = p.size * pr.scale * (1 + trebleEnergy * 2);
        const col = voiceColors[p.voice];
        const pAlpha = Math.min(1, pr.scale * 1.5) * (0.3 + bassEnergy * 0.6);
        ctx.fillStyle = `rgba(${col.rgb[0]}, ${col.rgb[1]}, ${col.rgb[2]}, ${pAlpha})`;
        ctx.beginPath();
        ctx.arc(pr.x, pr.y, Math.max(0.6, pSize), 0, Math.PI * 2);
        ctx.fill();
      }
    });
    ctx.restore();

    // --- G. REFINED BAROQUE FRAME & TYPOGRAPHY OVERLAY ---
    ctx.restore(); // Restore shake

    // Perimeter Filigree Baroque Border
    ctx.lineWidth = 2.0;
    ctx.strokeStyle = isPicardy ? "rgba(255, 215, 80, 0.75)" : "rgba(180, 160, 240, 0.45)";
    ctx.strokeRect(36, 36, W - 72, H - 72);
    ctx.lineWidth = 1.0;
    ctx.strokeRect(46, 46, W - 92, H - 92);

    // Corner flourishes
    const cornerSize = 28;
    const corners = [
      [36, 36], [W - 36, 36], [36, H - 36], [W - 36, H - 36]
    ];
    corners.forEach(([cx, cy], idx) => {
      const sx = idx % 2 === 0 ? 1 : -1;
      const sy = idx < 2 === true ? 1 : -1;
      ctx.beginPath();
      ctx.moveTo(cx, cy + sy * cornerSize);
      ctx.lineTo(cx, cy);
      ctx.lineTo(cx + sx * cornerSize, cy);
      ctx.stroke();
    });

    // Monumental Baroque Header Typography
    ctx.textAlign = "center";
    ctx.fillStyle = isPicardy ? "#ffd54f" : "#f1e9ff";
    ctx.font = "600 24px Georgia, serif";
    ctx.letterSpacing = "6px";
    ctx.fillText("JOHANN SEBASTIAN BACH", W / 2, 88);

    ctx.fillStyle = isPicardy ? "#ffffff" : "rgba(220, 210, 255, 0.75)";
    ctx.font = "italic 300 17px Georgia, serif";
    ctx.letterSpacing = "4px";
    ctx.fillText("TOCCATA & CANON À 4 VOIX · BWV ∞", W / 2, 116);

    // Dynamic Harmonic State Badge
    let harmonicStatus = "D MINOR : TOCCATA INVOCATION";
    if (elapsed >= 3.42 && elapsed < 7.00) harmonicStatus = "TRIPLE FUGUE IN CONTRAPUNCTO";
    else if (elapsed >= 7.00 && elapsed < 10.45) harmonicStatus = "STRETTO & CIRCLE OF FIFTHS";
    else if (elapsed >= 10.45 && elapsed < 12.68) harmonicStatus = "CADENTIAL DOMINANT SWELL (A7)";
    else if (elapsed >= 12.68) harmonicStatus = "THE PICARDY THIRD : D MAJOR APOTHEOSIS";

    ctx.fillStyle = isPicardy ? "#ffd54f" : "#80d8ff";
    ctx.font = "600 13px -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif";
    ctx.letterSpacing = "3px";
    ctx.fillText(harmonicStatus, W / 2, 146);

    // Voice Activity Matrix HUD (Bottom Left)
    const voiceNames = ["SOPRANO", "ALTO", "TENOR", "BASS"];
    const hudY = H - 95;
    for (let v = 0; v < 4; v++) {
      const vx = 80 + v * 130;
      const vActive = activeNotes.some(n => n.v === v);
      const col = voiceColors[v];

      ctx.fillStyle = vActive ? col.main : "rgba(100, 90, 130, 0.4)";
      ctx.beginPath();
      ctx.arc(vx - 14, hudY, 5, 0, Math.PI * 2);
      ctx.fill();

      ctx.font = "600 11px -apple-system, BlinkMacSystemFont, sans-serif";
      ctx.textAlign = "left";
      ctx.letterSpacing = "1.5px";
      ctx.fillText(voiceNames[v], vx - 4, hudY + 4);
    }

    // Benchmark Time & Polyphonic Density HUD (Bottom Right)
    ctx.textAlign = "right";
    ctx.fillStyle = isPicardy ? "#ffd54f" : "rgba(210, 200, 250, 0.85)";
    ctx.font = "600 13px Georgia, serif";
    ctx.letterSpacing = "2px";
    ctx.fillText(`${elapsed.toFixed(2)}s / 15.00s · 4 VOICES`, W - 80, hudY + 4);

    // Fine Progress Rule Line at Bottom Edge
    ctx.fillStyle = "rgba(140, 120, 200, 0.25)";
    ctx.fillRect(48, H - 54, W - 96, 2);

    ctx.fillStyle = isPicardy ? "#ffd54f" : "#00e5ff";
    ctx.fillRect(48, H - 54, (W - 96) * progress, 2);

    // Lens Vignette Shading
    const vignette = ctx.createRadialGradient(W / 2, H / 2, H * 0.42, W / 2, H / 2, H * 0.72);
    vignette.addColorStop(0, "rgba(0,0,0,0)");
    vignette.addColorStop(1, "rgba(0,0,0,0.65)");
    ctx.fillStyle = vignette;
    ctx.fillRect(0, 0, W, H);
  }

  // Launch render loop
  render();

  // Return teardown function if called before harness 15.0s termination
  return () => {
    cancelAnimationFrame(animId);
  };
}
