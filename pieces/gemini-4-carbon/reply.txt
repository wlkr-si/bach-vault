    const noiseAmt = timbre === 0 ? 0.62 : (timbre === 1 ? 0.52 : 0.36);
    const lpAlpha = timbre === 0 ? 0.72 : (timbre === 1 ? 0.55 : 0.32);

    for (let i = 0; i < initLen; i++) {
      const u = i / initLen;
      const tri = u < pickPos ? (u / pickPos) : ((1 - u) / (1 - pickPos));
      const w = random() * 2 - 1;
      lpInit += lpAlpha * (w - lpInit);
      const val = (tri * 2 - 1) * (1 - noiseAmt) + lpInit * noiseAmt;
      ring[i] = val;
      mean += val;
    }
    mean /= initLen;
    for (let i = 0; i < initLen; i++) ring[i] -= mean;

    let writePos = 0;
    let prevX = 0;
    let prevAp = 0;
    // Subtle wooden soundboard body resonator state
    let bodyY1 = 0, bodyY2 = 0;
    const bodyOmega = (2 * Math.PI * (timbre === 0 ? 520 : 285)) / sr;
    const bodyR = 0.985;
    const bA1 = -2 * bodyR * Math.cos(bodyOmega);
    const bA2 = bodyR * bodyR;

    const attackSamp = Math.max(8, Math.floor(sr * 0.0022));
    let peak = 1e-6;

    for (let i = 0; i < durSamples; i++) {
      const idx0 = (writePos - N + initLen * 4) % initLen;
      const idx1 = (idx0 - 1 + initLen) % initLen;
      const lpSample = (1 - damp) * ring[idx0] + damp * ring[idx1];
      const apOut = apCoeff * lpSample + prevX - apCoeff * prevAp;
      prevX = lpSample;
      prevAp = apOut;

    const env = audioContext.createGain();
    const baseGain = (timbre === 0 ? 0.21 : (timbre === 1 ? 0.20 : 0.23)) * vel;
    const ringTail = timbre === 2 ? 0.14 : 0.095;
    const stopT = Math.min(pieceEndAudio, startT + Math.min(buf.duration - 0.01, durSec + ringTail));

    env.gain.setValueAtTime(baseGain, startT);
    env.gain.setValueAtTime(baseGain, Math.max(startT + 0.01, stopT - 0.065));
    env.gain.linearRampToValueAtTime(0.0001, Math.max(startT + 0.02, stopT));

    const panner = audioContext.createStereoPanner();
    panner.pan.setValueAtTime(panVal, startT);

    src.connect(filt);
    filt.connect(env);
    env.connect(panner);
    panner.connect(dryBus);
    panner.connect(reverbHighpass);
    if (timbre !== 2) panner.connect(delaySend);

    src.start(startT);
    src.stop(Math.max(startT + 0.03, stopT + 0.01));

    // Delicate Crystalline Glass-Celesta Harmonic on Soprano & Picardy Arpeggio
    if (crystalSparkle > 0) {
      const freq = midiToFreq(midi);
      const car = audioContext.createOscillator();
      const mod = audioContext.createOscillator();
      const modGain = audioContext.createGain();
      const cEnv = audioContext.createGain();

      car.type = "sine";
      car.frequency.setValueAtTime(freq * 2, startT);
      mod.type = "sine";
      mod.frequency.setValueAtTime(freq * 6.005, startT);

      modGain.gain.setValueAtTime(freq * 1.4 * crystalSparkle, startT);
      modGain.gain.exponentialRampToValueAtTime(0.01, startT + 0.12);

      const cPeak = 0.026 * vel * crystalSparkle;
      const cStop = Math.min(pieceEndAudio, startT + Math.min(0.45, durSec + 0.18));
      cEnv.gain.setValueAtTime(0.0001, startT);
      cEnv.gain.linearRampToValueAtTime(cPeak, startT + 0.004);
      cEnv.gain.exponentialRampToValueAtTime(0.0001, Math.max(startT + 0.03, cStop));

      mod.connect(modGain);
      modGain.connect(car.frequency);
      car.connect(cEnv);
      cEnv.connect(panner);

      car.start(startT);
      mod.start(startT);
      car.stop(Math.max(startT + 0.04, cStop + 0.01));
      mod.stop(Math.max(startT + 0.04, cStop + 0.01));
    }
  }

  // Schedule Singing Bowed Viol / Analog Cello-Horn (Voice 2)
  function scheduleBowedViol(midi, startT, durSec, vel, panVal) {
    const freq = midiToFreq(midi);
    const stopT = Math.min(pieceEndAudio, startT + durSec + 0.06);
    if (stopT <= startT + 0.02) return;

    const osc1 = audioContext.createOscillator();
    const osc2 = audioContext.createOscillator();
    const osc3 = audioContext.createOscillator();
    osc1.type = "sawtooth";
    osc2.type = "sawtooth";
    osc3.type = "triangle";
    osc1.frequency.setValueAtTime(freq, startT);
    osc2.frequency.setValueAtTime(freq * 1.0032, startT); // +5.5 cents warm ensemble chorus
    osc3.frequency.setValueAtTime(freq * 0.9972, startT);

    // Delayed expressive human vibrato on sustained notes
    if (durSec > 0.26) {
      const vib = audioContext.createOscillator();
      const vibGain = audioContext.createGain();
      vib.frequency.setValueAtTime(5.5, startT);
      vibGain.gain.setValueAtTime(0.0001, startT);
      vibGain.gain.linearRampToValueAtTime(freq * 0.0048, startT + Math.min(durSec * 0.65, 0.24));
      vib.connect(vibGain);
      vibGain.connect(osc1.frequency);
      vibGain.connect(osc2.frequency);
      vib.start(startT);
      vib.stop(stopT + 0.02);
    }

    // Bow rosin articulation noise burst
    const rosin = audioContext.createBufferSource();
    rosin.buffer = noiseBuf;
    const rOffset = (midi * 0.037) % 1.0;
    const rosinBp = audioContext.createBiquadFilter();
    rosinBp.type = "bandpass";
    rosinBp.frequency.setValueAtTime(Math.min(4200, freq * 3.2), startT);
    rosinBp.Q.value = 2.5;
    const rosinEnv = audioContext.createGain();
    rosinEnv.gain.setValueAtTime(0.032 * vel, startT);
    rosinEnv.gain.exponentialRampToValueAtTime(0.0001, startT + 0.055);

    // Warm Cello Body Formant + Dynamic Bow Filter
    const bodyPeak = audioContext.createBiquadFilter();
    bodyPeak.type = "peaking";
    bodyPeak.frequency.value = 540;
    bodyPeak.Q.value = 1.6;
    bodyPeak.gain.value = 4.0;

    const lp = audioContext.createBiquadFilter();
    lp.type = "lowpass";
    lp.Q.value = 0.9;
    const fPeak = Math.min(3600, freq * (4.2 + 3.0 * vel));
    lp.frequency.setValueAtTime(freq * 1.8, startT);
    lp.frequency.exponentialRampToValueAtTime(fPeak, startT + Math.min(0.07, durSec * 0.35));
    lp.frequency.exponentialRampToValueAtTime(Math.max(260, fPeak * 0.68), stopT);

    const env = audioContext.createGain();
    const gPeak = 0.088 * vel;
    const atk = Math.min(0.038, durSec * 0.28);
    env.gain.setValueAtTime(0.0001, startT);
    env.gain.linearRampToValueAtTime(gPeak, startT + atk);
    env.gain.setValueAtTime(gPeak * 0.88, Math.max(startT + atk + 0.005, stopT - 0.065));
    env.gain.linearRampToValueAtTime(0.0001, stopT);

    const panner = audioContext.createStereoPanner();
    panner.pan.setValueAtTime(panVal, startT);

    osc1.connect(bodyPeak);
    osc2.connect(bodyPeak);
    osc3.connect(bodyPeak);
    rosin.connect(rosinBp);
    rosinBp.connect(rosinEnv);
    rosinEnv.connect(bodyPeak);

    bodyPeak.connect(lp);
    lp.connect(env);
    env.connect(panner);
    panner.connect(dryBus);
    panner.connect(reverbHighpass);

    osc1.start(startT);
    osc2.start(startT);
    osc3.start(startT);
    rosin.start(startT, rOffset, 0.08);
    osc1.stop(stopT + 0.02);
    osc2.stop(stopT + 0.02);
    osc3.stop(stopT + 0.02);
  }

  // Schedule 32' Cathedral Pipe Pedal (Voice 3)
  function schedulePedalOrgan(midi, startT, durSec, vel) {
    const freq = midiToFreq(midi); // e.g., D2 = 73.42Hz, A1 = 55Hz
    const stopT = Math.min(pieceEndAudio, startT + durSec + 0.08);
    if (stopT <= startT + 0.03) return;

    // 16' Principal + Sub-Bourdon + Octave Principal + Baroque Quint (1.498x)
    const oFund = audioContext.createOscillator();
    const oOct = audioContext.createOscillator();
    const oQuint = audioContext.createOscillator();
    oFund.type = "sine";
    oOct.type = "sawtooth";
    oQuint.type = "triangle";

    oFund.frequency.setValueAtTime(freq, startT);
    oOct.frequency.setValueAtTime(freq * 2, startT);
    oQuint.frequency.setValueAtTime(freq * 2.9966, startT); // 12th stop for unmistakable pitch definition

    const gFund = audioContext.createGain();
    const gOct = audioContext.createGain();
    const gQuint = audioContext.createGain();
    gFund.gain.value = 0.20 * vel;
    gOct.gain.value = 0.075 * vel;
    gQuint.gain.value = 0.028 * vel;

    const lp = audioContext.createBiquadFilter();
    lp.type = "lowpass";
    lp.frequency.setValueAtTime(Math.min(950, freq * 8.5), startT);
    lp.Q.value = 0.7;

    const env = audioContext.createGain();
    env.gain.setValueAtTime(0.0001, startT);
    env.gain.linearRampToValueAtTime(1.0, startT + 0.042);
    env.gain.setValueAtTime(0.92, Math.max(startT + 0.05, stopT - 0.09));
    env.gain.linearRampToValueAtTime(0.0001, stopT);

    oFund.connect(gFund);
    oOct.connect(gOct);
    oQuint.connect(gQuint);
    gOct.connect(lp);
    gQuint.connect(lp);

    gFund.connect(env);
    lp.connect(env);
    env.connect(dryBus);
    env.connect(reverbHighpass);

    oFund.start(startT);
    oOct.start(startT);
    oQuint.start(startT);
    oFund.stop(stopT + 0.02);
    oOct.stop(stopT + 0.02);
    oQuint.stop(stopT + 0.02);
  }

  // Schedule Modal Kettle Timpani
  function scheduleTimpani(midi, startT, vel) {
    const freq = midiToFreq(midi);
    const stopT = Math.min(pieceEndAudio, startT + 0.95);
    if (stopT <= startT + 0.03) return;

    // Drumhead modal partials (1.0, 1.504, 1.984) with initial membrane tension pitch drop
    const ratios = [1.0, 1.504, 1.984];
    const gains = [0.22, 0.065, 0.028];
    const decays = [0.78, 0.34, 0.18];

    for (let m = 0; m < 3; m++) {
      const osc = audioContext.createOscillator();
      const g = audioContext.createGain();
      osc.type = "sine";
      const fTarget = freq * ratios[m];
      osc.frequency.setValueAtTime(fTarget * 1.075, startT);
      osc.frequency.exponentialRampToValueAtTime(fTarget, startT + 0.055);

      const durM = Math.min(stopT - startT, decays[m]);
      g.gain.setValueAtTime(0.0001, startT);
      g.gain.linearRampToValueAtTime(gains[m] * vel, startT + 0.005);
      g.gain.exponentialRampToValueAtTime(0.0001, startT + Math.max(0.04, durM));

      osc.connect(g);
      g.connect(dryBus);
      if (m > 0) g.connect(reverbHighpass);
      osc.start(startT);
      osc.stop(startT + durM + 0.02);
    }

    // Felt mallet skin impact
    const mallet = audioContext.createBufferSource();
    mallet.buffer = noiseBuf;
    const bp = audioContext.createBiquadFilter();
    bp.type = "lowpass";
    bp.frequency.setValueAtTime(420 + 280 * vel, startT);
    const mg = audioContext.createGain();
    mg.gain.setValueAtTime(0.09 * vel, startT);
    mg.gain.exponentialRampToValueAtTime(0.0001, startT + 0.035);

    mallet.connect(bp);
    bp.connect(mg);
    mg.connect(dryBus);
    mallet.start(startT, 0.2, 0.05);
  }

  // Schedule Warm Brass / Pipe Principal Chorus (Cadential Suspension & D Major Picardy Chord)
  function scheduleBrassChord(midis, startT, durSec, vel, brightMajor) {
    const stopT = Math.min(pieceEndAudio, startT + durSec);
    if (stopT <= startT + 0.05) return;

    midis.forEach((midi, idx) => {
      const freq = midiToFreq(midi);
      const panVal = ((idx / Math.max(1, midis.length - 1)) * 2 - 1) * 0.58;

      const o1 = audioContext.createOscillator();
      const o2 = audioContext.createOscillator();
      o1.type = "sawtooth";
      o2.type = brightMajor ? "sawtooth" : "triangle";
      o1.frequency.setValueAtTime(freq * 0.9985, startT);
      o2.frequency.setValueAtTime(freq * 1.0022, startT);

      const lp = audioContext.createBiquadFilter();
      lp.type = "lowpass";
      lp.Q.value = 1.1;
      const targetCut = brightMajor ? Math.min(4400, freq * 6.5) : Math.min(2400, freq * 4.5);
      lp.frequency.setValueAtTime(freq * 1.5, startT);
      lp.frequency.exponentialRampToValueAtTime(targetCut, startT + Math.min(0.22, durSec * 0.3));
      lp.frequency.exponentialRampToValueAtTime(Math.max(240, targetCut * 0.55), stopT);

      const env = audioContext.createGain();
      const voiceGain = (brightMajor ? 0.044 : 0.032) * vel;
      env.gain.setValueAtTime(0.0001, startT);
      env.gain.linearRampToValueAtTime(voiceGain, startT + (brightMajor ? 0.055 : 0.075));
      env.gain.setValueAtTime(voiceGain * 0.85, Math.max(startT + 0.09, stopT - (brightMajor ? 0.45 : 0.08)));
      env.gain.exponentialRampToValueAtTime(0.0001, stopT);

      const panner = audioContext.createStereoPanner();
      panner.pan.setValueAtTime(panVal, startT);

      o1.connect(lp);
      o2.connect(lp);
      lp.connect(env);
      env.connect(panner);
      panner.connect(dryBus);
      panner.connect(reverbHighpass);

      o1.start(startT);
      o2.start(startT);
      o1.stop(stopT + 0.02);
      o2.stop(stopT + 0.02);
    });
  }

  // Schedule Horological Escapement Ticks (delicate precision clockwork in Acts I-II)
  function scheduleEscapementTick(startT, accent, panVal) {
    if (startT >= pieceEndAudio - 0.05) return;
    const src = audioContext.createBufferSource();
    src.buffer = noiseBuf;

    const hp = audioContext.createBiquadFilter();
    hp.type = "bandpass";
    hp.frequency.setValueAtTime(accent ? 4400 : 6200, startT);
    hp.Q.value = 5.5;

    const g = audioContext.createGain();
    const amp = accent ? 0.026 : 0.013;
    g.gain.setValueAtTime(amp, startT);
    g.gain.exponentialRampToValueAtTime(0.0001, startT + 0.012);

    const pan = audioContext.createStereoPanner();
    pan.pan.setValueAtTime(panVal, startT);

    src.connect(hp);
    hp.connect(g);
    g.connect(pan);
    pan.connect(dryBus);

    src.start(startT, 0.45, 0.02);
  }

  // Trigger full score onto Web Audio timeline
  for (const n of allNotes) {
    const startT = t0 + n.startSec;
    if (startT >= pieceEndAudio - 0.04) continue;
    if (n.voice === 0) {
      const isCoda = n.step >= 128;
      const sparkle = isCoda ? 1.0 : (n.vel > 0.90 ? 0.55 : 0.25);
      schedulePluck(n.midi, startT, n.durSec, n.vel, 0, 0.27, sparkle);
    } else if (n.voice === 1) {
      const isPizz = n.step < 16;
      schedulePluck(n.midi, startT, n.durSec, n.vel, isPizz ? 2 : 1, isPizz ? -0.10 : -0.31, n.step >= 128 ? 0.65 : 0);
    } else if (n.voice === 2) {
      scheduleBowedViol(n.midi, startT, n.durSec, n.vel, 0.12);
    } else if (n.voice === 3) {
      if (n.vel < 0.55) {
        schedulePluck(n.midi, startT, n.durSec, n.vel * 1.45, 2, -0.14, 0);
      } else {
        schedulePedalOrgan(n.midi, startT, n.durSec, n.vel);
      }
    }
  }

  for (const tp of timpaniEvents) {
    scheduleTimpani(tp.midi, t0 + tp.startSec, tp.vel);
  }

  // Grand Cadential Brass Chorus: Asus4 (Step 120) -> A7 (Step 124) -> Sunlit D Major Picardy (Step 128)
  scheduleBrassChord([52, 57, 62, 67, 69], t0 + stepToSec(120), stepToSec(124) - stepToSec(120) + 0.04, 0.85, false);
  scheduleBrassChord([52, 57, 61, 67, 69], t0 + stepToSec(124), stepToSec(128) - stepToSec(124) + 0.04, 0.92, false);
  const codaDur = Math.max(0.4, (duration - LEAD_IN - 0.06) - PICARDY_SEC);
  scheduleBrassChord([38, 45, 50, 57, 62, 66, 69, 74], t0 + PICARDY_SEC, codaDur, 1.0, true);
  // Final radiant high F#6 crystalline harmonic shimmer in Coda
  schedulePluck(90, t0 + stepToSec(132.5), 0.75 * timeScale, 0.58, 0, 0.0, 1.15);

  // Subtle clockwork escapement ticks in Bars 1-4 (gradually yielding to full orchestra)
  for (let s = 0; s < 64; s++) {
    const fade = s < 32 ? 1.0 : 1.0 - ((s - 32) / 32) * 0.65;
    if (s % 2 === 0 || s < 32) {
      const tickT = t0 + stepToSec(s);
      scheduleEscapementTick(tickT, (s % 4 === 0) && fade > 0.6, (s % 2 === 0 ? -0.22 : 0.22));
    }
  }

  // ============================================================================
  // 4. HYBRID WEBGL VOLUMETRIC CATHEDRAL SHADER + 2D PRECISION ARCHITECTURE
  // ============================================================================
  function createOffscreen(w, h) {
    if (typeof OffscreenCanvas !== "undefined") return new OffscreenCanvas(w, h);
    if (typeof document !== "undefined") {
      const c = document.createElement("canvas");
      c.width = w;
      c.height = h;
      return c;
    }
    return null;
  }

  const BG_W = Math.round(W * 0.5);
  const BG_H = Math.round(H * 0.5);
  const bgSurface = createOffscreen(BG_W, BG_H);
  let gl = null;
  let glProg = null;
  let glUniforms = null;

  if (bgSurface) {
    const glOpts = { alpha: false, depth: false, stencil: false, antialias: false, preserveDrawingBuffer: true };
    gl = bgSurface.getContext("webgl2", glOpts) || bgSurface.getContext("webgl", glOpts);
    if (gl) {
      const vsSrc = `
        attribute vec2 aPos;
        varying vec2 vUv;
        void main() {
          vUv = vec2(aPos.x * 0.5 + 0.5, 0.5 - aPos.y * 0.5);
          gl_Position = vec4(aPos, 0.0, 1.0);
        }
      `;
      const fsSrc = `
        precision highp float;
        varying vec2 vUv;
        uniform float uTime;
        uniform float uStep;
        uniform float uPicardy;
        uniform float uTension;
        uniform float uEnergy;
        uniform vec2 uRes;

        void main() {
          vec2 uv = vUv;
          float aspect = uRes.x / uRes.y;

          // Center aligned with Astrarium at (0.5, 0.4037)
          vec2 c = vec2(uv.x - 0.5, (uv.y - 0.4037) / aspect);
          float r = length(c);
          float ang = atan(c.y, c.x);

          // Rose-window 12 & 24-fold volumetric cathedral shafts
          float rot = uStep * 0.018;
          float rays12 = 0.5 + 0.5 * cos(ang * 12.0 - rot);
          float rays24 = 0.5 + 0.5 * cos(ang * 24.0 + rot * 1.5);
          float shaft = pow(rays12, 3.0) * 0.65 + pow(rays24, 5.0) * 0.35;
          float radialFalloff = exp(-r * 2.15) * (0.45 + 0.55 * smoothstep(0.02, 0.26, r));

          // Iterative optical caustics
          vec2 p = c * 4.2;
          float caus = 0.0;
          float t = uTime * 0.42;
          for (int i = 0; i < 3; i++) {
            float fi = float(i) + 1.0;
            p += vec2(
              sin(p.y * 1.7 + t * fi * 0.8) * 0.42,
              cos(p.x * 1.9 - t * fi * 0.7) * 0.42
            );
            caus += abs(sin(p.x + cos(p.y + t))) / fi;
          }
          caus = pow(max(0.0, 1.0 - caus * 0.46), 2.4);

          // Gothic Lancet Cathedral Niche Relief
          float archX = abs(uv.x - 0.5) * 2.0;
          float archCurve = archX * archX * 0.24 + max(0.0, 0.14 - uv.y) * 2.2;
          float nicheMask = 1.0 - smoothstep(0.72, 1.08, archX + archCurve * 0.45);

          // Nocturnal D Minor vs Tension vs Sunlit D Major Picardy Palette
          vec3 abyssMin = vec3(0.022, 0.034, 0.072);
          vec3 midMin   = vec3(0.055, 0.105, 0.195);
          vec3 rayMin   = vec3(0.32, 0.56, 0.88);
          vec3 warmBr   = vec3(0.88, 0.62, 0.28);

          vec3 tensCol  = vec3(0.68, 0.22, 0.36);
          vec3 goldAb   = vec3(0.085, 0.058, 0.028);
          vec3 goldMid  = vec3(0.32, 0.20, 0.07);
          vec3 goldRay  = vec3(1.00, 0.82, 0.42);

          vec3 baseCol = mix(abyssMin, goldAb, uPicardy * 0.85);
          vec3 vaultCol = mix(midMin, goldMid, uPicardy);
          vec3 shaftCol = mix(mix(rayMin, tensCol, uTension * 0.52), goldRay, uPicardy);

          vec3 col = mix(baseCol, vaultCol, nicheMask * exp(-r * 1.35));
          col += shaftCol * shaft * radialFalloff * (0.28 + 0.42 * uEnergy + 0.55 * uPicardy);
          col += mix(warmBr, goldRay, uPicardy) * caus * nicheMask * (0.09 + 0.14 * uEnergy + 0.18 * uPicardy);

          // Lower Mercury Basin ambient glow around y = 0.863
          vec2 bc = vec2((uv.x - 0.5) * 1.15, (uv.y - 0.863) * 4.8);
          float bDist = length(bc);
          float basinGlow = exp(-bDist * bDist * 1.8);
          col += mix(vec3(0.12, 0.25, 0.42), vec3(0.65, 0.45, 0.16), uPicardy) * basinGlow * (0.35 + 0.35 * uEnergy);

          // Subtle stone border & filmic tone map
          float vig = smoothstep(1.22, 0.28, length(vec2((uv.x - 0.5) * 1.55, (uv.y - 0.48) * 1.35)));
          col *= 0.52 + 0.48 * vig;
          col = col / (1.0 + col * 0.42);

          gl_FragColor = vec4(col, 1.0);
        }
      `;

      const compile = (type, src) => {
        const sh = gl.createShader(type);
        gl.shaderSource(sh, src);
        gl.compileShader(sh);
        if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS)) {
          gl.deleteShader(sh);
          return null;
        }
        return sh;
      };
      const vs = compile(gl.VERTEX_SHADER, vsSrc);
      const fs = compile(gl.FRAGMENT_SHADER, fsSrc);
      if (vs && fs) {
        const prog = gl.createProgram();
        gl.attachShader(prog, vs);
        gl.attachShader(prog, fs);
        gl.linkProgram(prog);
        if (gl.getProgramParameter(prog, gl.LINK_STATUS)) {
          glProg = prog;
          gl.useProgram(prog);
          const vbo = gl.createBuffer();
          gl.bindBuffer(gl.ARRAY_BUFFER, vbo);
          gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
          const loc = gl.getAttribLocation(prog, "aPos");
          gl.enableVertexAttribArray(loc);
          gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);
          glUniforms = {
            uTime: gl.getUniformLocation(prog, "uTime"),
            uStep: gl.getUniformLocation(prog, "uStep"),
            uPicardy: gl.getUniformLocation(prog, "uPicardy"),
            uTension: gl.getUniformLocation(prog, "uTension"),
            uEnergy: gl.getUniformLocation(prog, "uEnergy"),
            uRes: gl.getUniformLocation(prog, "uRes")
          };
        }
      }
    }
  }

  // ============================================================================
  // 5. KINETIC ASTRARIUM STATE: STRINGS, RINGS, PRISM, RIPPLES, MOTES
  // ============================================================================
  const CX = W * 0.5;
  const CY = H * 0.4037;      // 545 on 1350px
  const CROWN_Y = H * 0.118;  // 159 on 1350px
  const BASIN_Y = H * 0.863;  // 1165 on 1350px
  const BASIN_RX = W * 0.365; // 394px
  const BASIN_RY = H * 0.062; // 84px
  const FOV = 920;

  function project3D(x, y, z, cx = CX, cy = CY) {
    const s = FOV / Math.max(120, FOV + z);
    return { x: cx + x * s, y: cy + y * s, s, z };
  }

  function rotate3D(x, y, z, rx, ry, rz) {
    const cx = Math.cos(rx), sx = Math.sin(rx);
    let y1 = y * cx - z * sx;
    let z1 = y * sx + z * cx;
    const cy = Math.cos(ry), sy = Math.sin(ry);
    let x2 = x * cy + z1 * sy;
    let z2 = -x * sy + z1 * cy;
    const cz = Math.cos(rz), sz = Math.sin(rz);
    return {
      x: x2 * cz - y1 * sz,
      y: x2 * sz + y1 * cz,
      z: z2
    };

  function stringPoint(s, tNorm, vibOffset) {
    const inv = 1 - tNorm;
    const x = inv * inv * s.topX + 2 * inv * tNorm * s.midBowX + tNorm * tNorm * s.botX;
    const y = inv * inv * s.topY + 2 * inv * tNorm * CY + tNorm * tNorm * s.botY;
    return { x: x + vibOffset, y };

  let nextTimpCursor = 0;
  let picardyTriggered = false;

  let prismFlash = 0;

  // 6. MAIN RENDER LOOP (60fps Locked to audioContext.currentTime)

  function renderFrame() {
    const rawElapsed = audioContext.currentTime - t0;
    const elapsed = Math.max(0, Math.min(duration, rawElapsed));
    const dt = Math.max(0.001, Math.min(0.05, elapsed - prevElapsed || 0.016));
    prevElapsed = elapsed;

    const curStep = secToStep(elapsed);

    // Real-time audio RMS + Spectral readout
    analyser.getByteFrequencyData(freqData);
    analyser.getByteTimeDomainData(timeData);
    let sumSq = 0;
    for (let i = 0; i < timeData.length; i++) {
      const d = (timeData[i] - 128) / 128;
      sumSq += d * d;
    }
    const rms = Math.sqrt(sumSq / timeData.length);
    smoothEnergy += (Math.min(1, rms * 3.4) - smoothEnergy) * Math.min(1, dt * 12);

      if (curStep >= 84 && curStep < 88) tension = 1.0; // G#dim7 peak!
    }
    let picardy = 0;
    if (elapsed >= PICARDY_SEC) {
      picardy = Math.min(1, 1 - Math.exp(-(elapsed - PICARDY_SEC) * 5.2));
      if (!picardyTriggered) {
        picardyTriggered = true;
        prismFlash = 1.0;
        spawnShockwave(0.95, true);
        spawnShockwave(0.65, true);
      }
    }
    prismFlash *= Math.exp(-dt * 3.2);

      const ev = allNotes[nextNoteCursor++];
      const sIdx = midiToStringIndex(ev.midi);
      const st = strings[sIdx];
      st.amp = Math.min(20, st.amp * 0.35 + (6.5 + 11.5 * ev.vel));
      st.voice = ev.voice;
      st.vel = ev.vel;

          u: 0.44, // Starts at Astrarium intersection and shoots to Mercury Basin & Crown!
          dir: 1,
          vel: ev.vel,
          voice: ev.voice,
          isGold: ev.step >= 128
        });

      // Angular escapement impulse on corresponding voice ring
      rings[ev.voice].spinZ += (ev.voice % 2 === 0 ? 1 : -1) * (0.042 + 0.035 * ev.vel);
      prismFlash = Math.min(1.0, prismFlash + 0.14 * ev.vel);

    const activePitchClasses = new Float32Array(12);

      const n = allNotes[i];

      const tail = n.step >= 128 ? 0.65 : 0.06;
      if (n.endSec + tail >= elapsed) {
        const life = 1 - Math.max(0, Math.min(1, (elapsed - n.startSec) / (n.durSec + tail)));
        activeByVoice[n.voice].push({ midi: n.midi, vel: n.vel, life });
        // Pitch class with D = 0 (at 12 o'clock!)

        const nd = pcNodes[pc];
        const isTriad = (pc === 0 || pc === 4 || pc === 7); // D, F#, A
        ctx.strokeStyle = `rgba(255, 228, 145, ${((isTriad ? 0.52 : 0.18) * picardy).toFixed(3)})`;
        ctx.lineWidth = isTriad ? 2.4 : 1.0;
        ctx.beginPath();
        ctx.moveTo(CX, CY);
        ctx.lineTo(nd.x, nd.y);
        ctx.stroke();
      }
    }

    // Draw 12 Chromatic Pitch-Class Jewels around the Astrarium Dial
    for (let pc = 0; pc < 12; pc++) {
      const nd = pcNodes[pc];
      const act = nd.act;
      const rJewel = (pc === 0 ? 5.2 : 3.6) + 5.5 * act;
      if (act > 0.05) {
        const g = ctx.createRadialGradient(nd.x, nd.y, 1, nd.x, nd.y, rJewel * 3.4);
        g.addColorStop(0, `rgba(255, 250, 230, ${(0.85 * act).toFixed(3)})`);
        g.addColorStop(0.45, picardy > 0.3 ? `rgba(255, 205, 95, ${(0.45 * act).toFixed(3)})` : `rgba(120, 215, 255, ${(0.42 * act).toFixed(3)})`);
        g.addColorStop(1, "rgba(0,0,0,0)");
        ctx.fillStyle = g;
        ctx.beginPath();
        ctx.arc(nd.x, nd.y, rJewel * 3.4, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.fillStyle = act > 0.05 ? "#fffdf5" : `rgba(235, 198, 128, ${(0.38 + 0.35 * picardy).toFixed(3)})`;
      ctx.beginPath();
      ctx.arc(nd.x, nd.y, rJewel * 0.72, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    // --------------------------------------------------------------------------
    // LAYER 5: Kepler's Central 3D Stella Octangula Crystalline Prism
    // --------------------------------------------------------------------------
    const prismScale = (64 + 14 * smoothEnergy + 12 * prismFlash + 10 * picardy);
    const prx = elapsed * 0.48 + curStep * 0.015;
    const pry = elapsed * 0.62 - curStep * 0.02;
    const prz = curStep * 0.025;

    // Core Radiant Halo behind/inside Prism
    ctx.save();
    ctx.globalCompositeOperation = "lighter";
    const coreR = prismScale * (2.1 + 0.9 * picardy + 0.6 * prismFlash);
    const coreGrad = ctx.createRadialGradient(CX, CY, 4, CX, CY, coreR);
    coreGrad.addColorStop(0, "#ffffff");
    coreGrad.addColorStop(0.25, picardy > 0.25
      ? `rgba(255, 224, 132, ${(0.55 + 0.35 * picardy).toFixed(3)})`
      : `rgba(145, 220, 255, ${(0.38 + 0.32 * smoothEnergy).toFixed(3)})`);
    coreGrad.addColorStop(1, "rgba(0,0,0,0)");
    ctx.fillStyle = coreGrad;
    ctx.beginPath();
    ctx.arc(CX, CY, coreR, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    // Project & Depth-Sort all 8 Faces of Dual Tetrahedra
    const facesToDraw = [];
    const buildTetFaces = (verts, isDual) => {
      const p3s = verts.map(([vx, vy, vz]) => rotate3D(vx * prismScale, vy * prismScale, vz * prismScale, prx, pry, prz));
      const p2s = p3s.map((p) => project3D(p.x, p.y, p.z));
      for (const [i0, i1, i2] of tetFaces) {
        const a = p3s[i0], b = p3s[i1], c = p3s[i2];
        const zAvg = (a.z + b.z + c.z) / 3;
        const ux = b.x - a.x, uy = b.y - a.y, uz = b.z - a.z;
        const vx = c.x - a.x, vy = c.y - a.y, vz = c.z - a.z;
        let nx = uy * vz - uz * vy;
        let ny = uz * vx - ux * vz;
        let nz = ux * vy - uy * vx;
        const nLen = Math.hypot(nx, ny, nz) || 1;
        nx /= nLen; ny /= nLen; nz /= nLen;
        // Directional cathedral light from top-front-left
        const lit = Math.max(0, -(nx * 0.35 + ny * 0.52 + nz * 0.78));
        facesToDraw.push({
          pts: [p2s[i0], p2s[i1], p2s[i2]],
          zAvg,
          lit,
          isDual
        });
      }
    };
    buildTetFaces(tetA, false);
    buildTetFaces(tetB, true);
    facesToDraw.sort((a, b) => b.zAvg - a.zAvg); // Painter's back-to-front

    ctx.save();
    for (const f of facesToDraw) {
      const alpha = 0.16 + 0.34 * f.lit + 0.14 * prismFlash;
      if (picardy > 0.35) {
        ctx.fillStyle = f.isDual
          ? `rgba(255, 232, 165, ${alpha.toFixed(3)})`
          : `rgba(255, 198, 96, ${alpha.toFixed(3)})`;
      } else {
        ctx.fillStyle = f.isDual
          ? `rgba(${Math.round(235 + 20 * tension)}, ${Math.round(182 - 40 * tension)}, 125, ${alpha.toFixed(3)})`
          : `rgba(125, 212, 255, ${alpha.toFixed(3)})`;
      }
      ctx.strokeStyle = `rgba(255, 248, 228, ${(0.38 + 0.48 * f.lit + 0.14 * picardy).toFixed(3)})`;
      ctx.lineWidth = 1.35;

      ctx.beginPath();
      ctx.moveTo(f.pts[0].x, f.pts[0].y);
      ctx.lineTo(f.pts[1].x, f.pts[1].y);
      ctx.lineTo(f.pts[2].x, f.pts[2].y);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
    }
    ctx.restore();

    // --------------------------------------------------------------------------
    // LAYER 6: Front Hemisphere of 3D Gimbaled Rings + Horological Escapement Anchor
    // --------------------------------------------------------------------------
    drawRingPass(true);

    // Rocking Clockwork Anchor Escapement Pallet at 12 o'clock of Ring 3
    ctx.save();
    const anchorY = CY - rings[3].radius - 16;
    const palletRock = Math.sin(curStep * Math.PI) * 0.22 * (1 - 0.75 * picardy);
    ctx.translate(CX, anchorY);
    ctx.rotate(palletRock);
    ctx.strokeStyle = `rgba(255, 224, 152, ${(0.72 + 0.25 * picardy).toFixed(3)})`;
    ctx.lineWidth = 2.4;
    ctx.beginPath();
    ctx.moveTo(0, -18);
    ctx.lineTo(0, 4);
    ctx.moveTo(-26, 13);
    ctx.quadraticCurveTo(0, -2, 26, 13);
    ctx.moveTo(-26, 13);
    ctx.lineTo(-22, 22);
    ctx.moveTo(26, 13);
    ctx.lineTo(22, 22);
    ctx.stroke();
    ctx.fillStyle = "#fff6db";
    ctx.beginPath();
    ctx.arc(0, -4, 3.6, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();

    // --------------------------------------------------------------------------
    // LAYER 7: 3D Perspective Mercury Acoustic Basin + FFT Mirror + Ripples
    // --------------------------------------------------------------------------
    ctx.save();
    // Carved dark obsidian & brass elliptical pool
    ctx.beginPath();
    ctx.ellipse(CX, BASIN_Y, BASIN_RX, BASIN_RY, 0, 0, Math.PI * 2);
    const poolGrad = ctx.createRadialGradient(CX, BASIN_Y - 12, 18, CX, BASIN_Y, BASIN_RX);
    poolGrad.addColorStop(0, picardy > 0.35 ? "rgba(96, 68, 24, 0.92)" : "rgba(18, 36, 64, 0.92)");
    poolGrad.addColorStop(0.68, picardy > 0.35 ? "rgba(32, 22, 10, 0.95)" : "rgba(8, 15, 28, 0.95)");
    poolGrad.addColorStop(1, "rgba(4, 7, 14, 0.98)");
    ctx.fillStyle = poolGrad;
    ctx.fill();

    // Clip interior of Mercury Basin for FFT spectral reflections & liquid ripples
    ctx.save();
    ctx.clip();
    ctx.globalCompositeOperation = "lighter";

    // Real-Time Mirrored Spectral Columns shimmering in the liquid mercury
    const BARS = 44;
    for (let b = 0; b < BARS; b++) {
      const sym = Math.abs(b - (BARS - 1) * 0.5) / (BARS * 0.5);
      const bin = Math.min(freqData.length - 1, Math.floor(2 + sym * 58));
      const mag = (freqData[bin] / 255) * (1 - 0.35 * sym);
      if (mag < 0.04) continue;
      const bx = CX + ((b / (BARS - 1)) * 2 - 1) * (BASIN_RX * 0.84);
      const bh = mag * BASIN_RY * 1.45;
      const colGrad = ctx.createLinearGradient(bx, BASIN_Y - bh * 0.35, bx, BASIN_Y + bh * 0.75);
      colGrad.addColorStop(0, "rgba(0,0,0,0)");
      colGrad.addColorStop(0.45, picardy > 0.3
        ? `rgba(255, 218, 125, ${(0.34 * mag).toFixed(3)})`
        : `rgba(135, 212, 255, ${(0.28 * mag).toFixed(3)})`);
      colGrad.addColorStop(1, "rgba(0,0,0,0)");
      ctx.strokeStyle = colGrad;
      ctx.lineWidth = 4.5;
      ctx.beginPath();
      ctx.moveTo(bx, BASIN_Y - bh * 0.35);
      ctx.lineTo(bx, BASIN_Y + bh * 0.75);
      ctx.stroke();
    }

    // Perspective-Projected Concentric Liquid Interference Ripples
    for (let i = ripples.length - 1; i >= 0; i--) {
      const rp = ripples[i];
      rp.r += dt * 135;
      const prog = rp.r / rp.maxR;
      if (prog >= 1) {
        ripples.splice(i, 1);
        continue;
      }
      const a = rp.alpha * (1 - prog) * (1 - prog);
      ctx.strokeStyle = rp.isGold || picardy > 0.3
        ? `rgba(255, 226, 138, ${a.toFixed(3)})`
        : voiceRGBA(rp.voice, a, picardy);
      ctx.lineWidth = 1.9 * (1 - 0.5 * prog);
      ctx.beginPath();
      ctx.ellipse(rp.x, rp.y, rp.r, rp.r * (BASIN_RY / BASIN_RX), 0, 0, Math.PI * 2);
      ctx.stroke();
    }
    ctx.restore();

    // Engraved Double Brass Rim around Mercury Basin
    ctx.strokeStyle = `rgba(242, 202, 128, ${(0.55 + 0.35 * picardy).toFixed(3)})`;
    ctx.lineWidth = 2.6;
    ctx.beginPath();
    ctx.ellipse(CX, BASIN_Y, BASIN_RX, BASIN_RY, 0, 0, Math.PI * 2);
    ctx.stroke();

    ctx.strokeStyle = `rgba(242, 202, 128, ${(0.22 + 0.20 * picardy).toFixed(3)})`;
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    ctx.ellipse(CX, BASIN_Y + 6, BASIN_RX + 10, BASIN_RY + 5, 0, 0, Math.PI * 2);
    ctx.stroke();
    ctx.restore();

    // --------------------------------------------------------------------------
    // LAYER 8: Expanding Harmonic Shockwaves & Floating Gold-Leaf Dust Motes
    // --------------------------------------------------------------------------
    ctx.save();
    ctx.globalCompositeOperation = "lighter";

    for (let i = shockwaves.length - 1; i >= 0; i--) {
      const sw = shockwaves[i];
      sw.r += dt * (sw.isPicardy ? 540 : 360);
      const prog = sw.r / sw.maxR;
      if (prog >= 1) {
        shockwaves.splice(i, 1);
        continue;
      }
      const a = sw.alpha * Math.pow(1 - prog, 1.8);
      ctx.strokeStyle = sw.isPicardy
        ? `rgba(255, 232, 152, ${a.toFixed(3)})`
        : `rgba(245, 198, 128, ${(a * 0.65).toFixed(3)})`;
      ctx.lineWidth = sw.isPicardy ? (6.0 * (1 - prog) + 1.2) : (3.0 * (1 - prog) + 0.8);
      ctx.beginPath();
      ctx.arc(CX, CY, sw.r, 0, Math.PI * 2);
      ctx.stroke();
    }

    const speedBoost = 1 + 1.6 * smoothEnergy + 0.8 * tension;
    for (let i = 0; i < NUM_MOTES; i++) {
      const m = motes[i];
      m.x += (m.vx + Math.sin(elapsed * 1.4 + m.phase) * 10) * dt * (1 - 0.4 * picardy);
      m.y += m.vy * dt * speedBoost * (1 - 0.35 * picardy);
      if (m.y < -10) m.y = H + 10;
      if (m.x < -10) m.x = W + 10;
      if (m.x > W + 10) m.x = -10;

      const tw = 0.45 + 0.55 * Math.sin(elapsed * 3.8 + m.phase);
      const alpha = (0.18 + 0.36 * tw) * (0.65 + 0.55 * picardy);
      ctx.fillStyle = picardy > 0.3 || m.hueType !== 0
        ? `rgba(255, 224, 148, ${alpha.toFixed(3)})`
        : `rgba(155, 224, 255, ${alpha.toFixed(3)})`;
      ctx.beginPath();
      ctx.arc(m.x, m.y, m.r * (0.8 + 0.4 * tw), 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();

    if (rawElapsed < duration + 0.5) {
      requestAnimationFrame(renderFrame);
    }
  }

  requestAnimationFrame(renderFrame);
}
