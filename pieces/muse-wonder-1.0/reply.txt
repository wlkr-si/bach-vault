      g.gain.setValueAtTime(v, t); g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
      s.connect(fl); fl.connect(g); s.start(t); s.stop(t + dur + 0.02);
    }
  }

    { n: "A",  b: 45, tones: [57, 61, 64, 69], pad: [52, 57, 61, 64], pal: [[44,21,51],[102,48,75],[29,17,39]], acc: [251,113,133] },
