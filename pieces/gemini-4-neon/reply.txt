export const title = 'Nocturne of the Brass Astrolabe';

export function start({ audioContext, canvas, duration = 15, random }) {
  // Deterministic fallback PRNG if random is not provided
  const rand = typeof random === 'function' ? random : (() => {
    let s = 0x1a2b3c4d;
    return () => {
      s |= 0;
      s = (s + 0x6d2b79f5) | 0;
      let t = Math.imul(s ^ (s >>> 15), 1 | s);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  })();

  const W = 1080;
  const H = 1350;
  const CX = 540;
  const CY = 615;
  const TOTAL_DUR = Math.max(1, duration || 15);
  const TIME_SCALE = TOTAL_DUR / 15;
  const BPM = 128;
  const BEAT = (60 / BPM) * TIME_SCALE; // 32 beats = 15.000s exact

  if (audioContext && audioContext.state === 'suspended') {
    audioContext.resume().catch(() => {});
  }

  // =========================================================================
  // 1. MASTER AUDIO ENGINE & STUDIO EFFECTS BUS
  // =========================================================================
  const ctx = audioContext;
  const now0 = ctx.currentTime;
  const startTime = now0 + 0.045;

  const midiToFreq = (m) => 440 * Math.pow(2, (m - 69) / 12);

  const createPanner = (panVal) => {
    if (typeof ctx.createStereoPanner === 'function') {
      const p = ctx.createStereoPanner();
      p.pan.value = Math.max(-1, Math.min(1, panVal));
      return p;
    }
    return ctx.createGain();
  };

  // Master signal chain: mixBus -> warmthEQ -> softClip -> compressor -> masterFade -> analyser -> destination
  const mixBus = ctx.createGain();
  mixBus.gain.value = 0.78;

  const lowShelf = ctx.createBiquadFilter();
  lowShelf.type = 'lowshelf';
  lowShelf.frequency.value = 95;
  lowShelf.gain.value = 2.2;

  const mudDip = ctx.createBiquadFilter();
  mudDip.type = 'peaking';
  mudDip.frequency.value = 310;
  mudDip.Q.value = 1.1;
  mudDip.gain.value = -1.8;

  const airShelf = ctx.createBiquadFilter();
  airShelf.type = 'highshelf';
  airShelf.frequency.value = 6200;
  airShelf.gain.value = 2.6;

  // Gentle analog soft-clipper
  const saturator = ctx.createWaveShaper();
  const satCurve = new Float32Array(2048);
  const satNorm = Math.tanh(1.3);
  for (let i = 0; i < 2048; i++) {
    const x = (i / 1023.5) - 1;
    satCurve[i] = Math.tanh(x * 1.3) / satNorm;
  }
  saturator.curve = satCurve;
  saturator.oversample = '2x';

  const compressor = ctx.createDynamicsCompressor();
  compressor.threshold.value = -12;
  compressor.knee.value = 18;
  compressor.ratio.value = 3.2;
  compressor.attack.value = 0.012;
  compressor.release.value = 0.18;

  const masterFade = ctx.createGain();
  masterFade.gain.setValueAtTime(0.0001, now0);
  masterFade.gain.linearRampToValueAtTime(1.0, startTime + 0.03);
  const fadeStart = startTime + Math.max(0.5, TOTAL_DUR - 0.85);
  const fadeEnd = startTime + Math.max(0.6, TOTAL_DUR - 0.04);
  masterFade.gain.setValueAtTime(1.0, fadeStart);
  masterFade.gain.exponentialRampToValueAtTime(0.0001, fadeEnd);

  const analyser = ctx.createAnalyser();
  analyser.fftSize = 1024;
  analyser.smoothingTimeConstant = 0.8;

  mixBus.connect(lowShelf);
  lowShelf.connect(mudDip);
  mudDip.connect(airShelf);
  airShelf.connect(saturator);
  saturator.connect(compressor);
  compressor.connect(masterFade);
  masterFade.connect(analyser);
  analyser.connect(ctx.destination);

  // Algorithmic Cathedral Convolution Reverb (deterministic stereo IR)
  const buildCathedralIR = (seconds = 2.6) => {
    const sr = ctx.sampleRate;
    const len = Math.floor(sr * seconds);
    const buf = ctx.createBuffer(2, len, sr);
    const earlyTaps = [
      { t: 0.017, l: 0.55, r: 0.25 },
      { t: 0.029, l: 0.22, r: 0.52 },
      { t: 0.043, l: 0.40, r: 0.32 },
      { t: 0.061, l: 0.28, r: 0.38 },
      { t: 0.083, l: 0.22, r: 0.20 },
    ];
    for (let ch = 0; ch < 2; ch++) {
      const data = buf.getChannelData(ch);
      let lp = 0;
      const preDelay = Math.floor(sr * 0.019);
      for (let i = preDelay; i < len; i++) {
        const u = (i - preDelay) / (len - preDelay);
        const env = Math.pow(1 - u, 2.75) * Math.exp(-u * 2.8);
        const white = rand() * 2 - 1;
        // Progressive high-frequency air damping across the hall tail
        const coeff = 0.06 + 0.34 * Math.pow(1 - u, 1.4);
        lp += coeff * (white - lp);
        data[i] = lp * env * 0.58;
      }
      for (const tap of earlyTaps) {
        const idx = Math.floor(tap.t * sr);
        if (idx < len) {
          data[idx] += (ch === 0 ? tap.l : tap.r) * 0.35;
        }
      }
    }
    return buf;
  };

  const reverbSend = ctx.createGain();
  reverbSend.gain.value = 1.0;
  const convolver = ctx.createConvolver();
  convolver.buffer = buildCathedralIR(2.65);
  const reverbLP = ctx.createBiquadFilter();
  reverbLP.type = 'lowpass';
  reverbLP.frequency.value = 5200;
  const reverbReturn = ctx.createGain();
  reverbReturn.gain.value = 0.38;

  reverbSend.connect(convolver);
  convolver.connect(reverbLP);
  reverbLP.connect(reverbReturn);
  reverbReturn.connect(mixBus);

  // Stereo Ping-Pong Tape Delay for crystalline counterpoint echoes
  const delaySend = ctx.createGain();
  delaySend.gain.value = 1.0;
  const delayL = ctx.createDelay(1.0);
  const delayR = ctx.createDelay(1.0);
  delayL.delayTime.value = BEAT * 0.75; // dotted 8th
  delayR.delayTime.value = BEAT * 0.5;  // 8th
  const delayFilter = ctx.createBiquadFilter();
  delayFilter.type = 'lowpass';
  delayFilter.frequency.value = 3100;
  const delayFb = ctx.createGain();
  delayFb.gain.value = 0.26;
  const delayPanL = createPanner(-0.68);
  const delayPanR = createPanner(0.68);
  const delayReturn = ctx.createGain();
  delayReturn.gain.value = 0.21;

  delaySend.connect(delayFilter);
  delayFilter.connect(delayL);
  delayL.connect(delayR);
  delayR.connect(delayFb);
  delayFb.connect(delayFilter);
  delayL.connect(delayPanL);
  delayR.connect(delayPanR);
  delayPanL.connect(delayReturn);
  delayPanR.connect(delayReturn);
  delayReturn.connect(mixBus);

  // Short shared white-noise buffer for organic felt-hammer & clockwork clicks
  const noiseBuf = ctx.createBuffer(1, Math.floor(ctx.sampleRate * 0.25), ctx.sampleRate);
  {
    const ch = noiseBuf.getChannelData(0);
    for (let i = 0; i < ch.length; i++) ch[i] = rand() * 2 - 1;
  }

  // =========================================================================
  // 2. BESPOKE INSTRUMENT SYNTHESIS
  // =========================================================================

  // Instrument A: Harp / Celesta / Felt-Piano Hybrid (Bachian Counterpoint)
  const playHarpCelesta = (t, midi, vel = 0.5, dur = 1.2, pan = 0) => {
    const freq = midiToFreq(midi);
    const out = ctx.createGain();
    const panner = createPanner(pan);
    out.connect(panner);
    panner.connect(mixBus);

    const revTap = ctx.createGain();
    revTap.gain.value = 0.42;
    panner.connect(revTap);
    revTap.connect(reverbSend);

    const delTap = ctx.createGain();
    delTap.gain.value = midi >= 64 ? 0.26 : 0.12;
    panner.connect(delTap);
    delTap.connect(delaySend);

    // Filter envelope
    const filter = ctx.createBiquadFilter();
    filter.type = 'lowpass';
    const peakCut = Math.min(12000, freq * (4.2 + vel * 4.5));
    const tailCut = Math.min(6000, freq * 1.9);
    filter.frequency.setValueAtTime(peakCut, t);
    filter.frequency.exponentialRampToValueAtTime(Math.max(180, tailCut), t + dur * 0.85);
    filter.Q.value = 0.85;
    filter.connect(out);

    // Core warm body (triangle + subtle 2nd harmonic sine)
    const osc1 = ctx.createOscillator();
    osc1.type = 'triangle';
    osc1.frequency.setValueAtTime(freq, t);

    const osc2 = ctx.createOscillator();
    osc2.type = 'sine';
    osc2.frequency.setValueAtTime(freq * 2, t);
    const g2 = ctx.createGain();
    g2.gain.setValueAtTime(0.24 * vel, t);
    g2.gain.exponentialRampToValueAtTime(0.0001, t + dur * 0.55);

    // FM glass/celesta bell transient
    const fmCarrier = ctx.createOscillator();
    const fmMod = ctx.createOscillator();
    const fmDepth = ctx.createGain();
    const fmOut = ctx.createGain();
    fmCarrier.type = 'sine';
    fmMod.type = 'sine';
    fmCarrier.frequency.setValueAtTime(freq, t);
    fmMod.frequency.setValueAtTime(freq * 3.003, t);
    fmDepth.gain.setValueAtTime(freq * (0.9 + vel * 1.1), t);
    fmDepth.gain.exponentialRampToValueAtTime(0.1, t + 0.11);
    fmOut.gain.setValueAtTime(0.22 * vel, t);
    fmOut.gain.exponentialRampToValueAtTime(0.0001, t + Math.min(dur, 0.75));
    fmMod.connect(fmDepth);
    fmDepth.connect(fmCarrier.frequency);
    fmCarrier.connect(fmOut);
    fmOut.connect(filter);

    // Subtle wooden hammer attack transient
    const hammer = ctx.createBufferSource();
    hammer.buffer = noiseBuf;
    const hFilt = ctx.createBiquadFilter();
    hFilt.type = 'bandpass';
    hFilt.frequency.value = Math.min(5500, freq * 2.4);
    hFilt.Q.value = 2.0;
    const hGain = ctx.createGain();
    hGain.gain.setValueAtTime(0.065 * vel, t);
    hGain.gain.exponentialRampToValueAtTime(0.0001, t + 0.018);
    hammer.connect(hFilt);
    hFilt.connect(hGain);
    hGain.connect(out);

    // Amplitude envelope
    const peak = 0.25 * vel;
    out.gain.setValueAtTime(0.0001, t);
    out.gain.linearRampToValueAtTime(peak, t + 0.006);
    out.gain.exponentialRampToValueAtTime(Math.max(0.0002, peak * 0.34), t + Math.min(0.24, dur * 0.35));
    out.gain.exponentialRampToValueAtTime(0.0001, t + dur);

    osc1.connect(filter);
    osc2.connect(g2);
    g2.connect(filter);

    const stopT = t + dur + 0.05;
    osc1.start(t);
    osc2.start(t);
    fmCarrier.start(t);
    fmMod.start(t);
    hammer.start(t);
    osc1.stop(stopT);
    osc2.stop(stopT);
    fmCarrier.stop(stopT);
    fmMod.stop(stopT);
    hammer.stop(t + 0.03);
  };

  // Instrument B: Soaring Cantus Lead (Bowed Crystalline Chamber Voice)
  const playCantusVoice = (t, midi, vel = 0.55, dur = 1.0, pan = 0, octaveDouble = false) => {
    const freq = midiToFreq(midi);
    const out = ctx.createGain();
    const panner = createPanner(pan);
    out.connect(panner);
    panner.connect(mixBus);

    const revTap = ctx.createGain();
    revTap.gain.value = 0.62;
    panner.connect(revTap);
    revTap.connect(reverbSend);

    const delTap = ctx.createGain();
    delTap.gain.value = 0.30;
    panner.connect(delTap);
    delTap.connect(delaySend);

    const filter = ctx.createBiquadFilter();
    filter.type = 'lowpass';
    filter.Q.value = 1.4;
    const baseCut = Math.min(4800, freq * 2.1);
    const peakCut = Math.min(9000, freq * 4.8);
    filter.frequency.setValueAtTime(baseCut, t);
    filter.frequency.exponentialRampToValueAtTime(peakCut, t + Math.min(0.22, dur * 0.35));
    filter.frequency.exponentialRampToValueAtTime(Math.max(400, baseCut * 0.85), t + dur + 0.3);
    filter.connect(out);

    // Expressive vibrato LFO fading in gently
    const lfo = ctx.createOscillator();
    const lfoGain = ctx.createGain();
    lfo.frequency.value = 5.3;
    lfoGain.gain.setValueAtTime(0.0001, t);
    lfoGain.gain.linearRampToValueAtTime(freq * 0.0055, t + Math.min(0.32, dur * 0.5));
    lfo.connect(lfoGain);

    const oscs = [];
    const detunes = [-5, 5];
    for (const d of detunes) {
      const o = ctx.createOscillator();
      o.type = 'sawtooth';
      o.frequency.setValueAtTime(freq, t);
      o.detune.value = d;
      lfoGain.connect(o.frequency);
      const g = ctx.createGain();
      g.gain.value = 0.34;
      o.connect(g);
      g.connect(filter);
      oscs.push(o);
    }

    // Pure glass harmonic core
    const oPure = ctx.createOscillator();
    oPure.type = 'triangle';
    oPure.frequency.setValueAtTime(freq, t);
    lfoGain.connect(oPure.frequency);
    const gPure = ctx.createGain();
    gPure.gain.value = 0.55;
    oPure.connect(gPure);
    gPure.connect(filter);
    oscs.push(oPure);

    if (octaveDouble) {
      const oSub = ctx.createOscillator();
      oSub.type = 'triangle';
      oSub.frequency.setValueAtTime(freq * 0.5, t);
      const gSub = ctx.createGain();
      gSub.gain.value = 0.42;
      oSub.connect(gSub);
      gSub.connect(filter);
      oscs.push(oSub);
    }

    const peak = 0.16 * vel;
    const atk = Math.min(0.11, dur * 0.25);
    const rel = Math.min(0.65, dur * 0.55 + 0.2);
    out.gain.setValueAtTime(0.0001, t);
    out.gain.linearRampToValueAtTime(peak, t + atk);
    out.gain.setValueAtTime(peak * 0.88, t + Math.max(atk + 0.02, dur * 0.78));
    out.gain.exponentialRampToValueAtTime(0.0001, t + dur + rel);

    const stopT = t + dur + rel + 0.05;
    lfo.start(t);
    lfo.stop(stopT);
    for (const o of oscs) {
      o.start(t);
      o.stop(stopT);
    }
  };

  // Instrument C: Lush Chamber Strings / Cathedral Organ Pad
  const playChamberPad = (t, midis, vel = 0.38, dur = 2.0, bright = 1.0) => {
    const out = ctx.createGain();
    const filter = ctx.createBiquadFilter();
    filter.type = 'lowpass';
    filter.Q.value = 0.7;
    const startCut = 340 * bright;
    const peakCut = Math.min(3400, 1150 * bright);
    filter.frequency.setValueAtTime(startCut, t);
    filter.frequency.exponentialRampToValueAtTime(peakCut, t + dur * 0.42);
    filter.frequency.exponentialRampToValueAtTime(Math.max(260, startCut * 1.1), t + dur + 0.65);

    filter.connect(out);
    out.connect(mixBus);

    const revTap = ctx.createGain();
    revTap.gain.value = 0.68;
    out.connect(revTap);
    revTap.connect(reverbSend);

    const atk = Math.min(0.45, dur * 0.28);
    const rel = Math.min(1.15, dur * 0.45 + 0.35);
    const peak = (0.145 * vel) / Math.sqrt(midis.length);
    out.gain.setValueAtTime(0.0001, t);
    out.gain.linearRampToValueAtTime(peak, t + atk);
    out.gain.setValueAtTime(peak * 0.92, t + Math.max(atk + 0.05, dur * 0.82));
    out.gain.exponentialRampToValueAtTime(0.0001, t + dur + rel);

    const stopT = t + dur + rel + 0.05;
    midis.forEach((m, idx) => {
      const f = midiToFreq(m);
      const spread = midis.length > 1 ? ((idx / (midis.length - 1)) - 0.5) * 0.85 : 0;
      const p = createPanner(spread);
      p.connect(filter);
      for (const det of [-6.5, 6.5]) {
        const o = ctx.createOscillator();
        o.type = idx === 0 ? 'triangle' : 'sawtooth';
        o.frequency.setValueAtTime(f, t);
        o.detune.value = det + (rand() - 0.5) * 2;
        const g = ctx.createGain();
        g.gain.value = idx === 0 ? 0.62 : 0.42;
        o.connect(g);
        g.connect(p);
        o.start(t);
        o.stop(stopT);
      }
    });
  };

  // Instrument D: Deep Cathedral Organ / Cello Pedal Bass
  const playPedalBass = (t, midi, vel = 0.65, dur = 2.0) => {
    const freq = midiToFreq(midi);
    const out = ctx.createGain();
    out.connect(mixBus);

    const revTap = ctx.createGain();
    revTap.gain.value = 0.22;
    out.connect(revTap);
    revTap.connect(reverbSend);

    const sub = ctx.createOscillator();
    sub.type = 'sine';
    sub.frequency.setValueAtTime(freq, t);

    const bow = ctx.createOscillator();
    bow.type = 'sawtooth';
    bow.frequency.setValueAtTime(freq, t);

    const bow2 = ctx.createOscillator();
    bow2.type = 'triangle';
    bow2.frequency.setValueAtTime(freq * 2, t);

    const lp = ctx.createBiquadFilter();
    lp.type = 'lowpass';
    lp.frequency.setValueAtTime(Math.max(180, freq * 3.8), t);
    lp.frequency.exponentialRampToValueAtTime(Math.max(120, freq * 2.1), t + dur);
    lp.Q.value = 0.9;

    const gSub = ctx.createGain();
    gSub.gain.value = 0.72;
    const gBow = ctx.createGain();
    gBow.gain.value = 0.34;
    const gBow2 = ctx.createGain();
    gBow2.gain.value = 0.16;

    sub.connect(gSub);
    gSub.connect(out);
    bow.connect(gBow);
    gBow.connect(lp);
    bow2.connect(gBow2);
    gBow2.connect(lp);
    lp.connect(out);

    const atk = Math.min(0.14, dur * 0.18);
    const rel = Math.min(0.85, dur * 0.4 + 0.25);
    const peak = 0.27 * vel;
    out.gain.setValueAtTime(0.0001, t);
    out.gain.linearRampToValueAtTime(peak, t + atk);
    out.gain.setValueAtTime(peak * 0.85, t + Math.max(atk + 0.05, dur * 0.85));
    out.gain.exponentialRampToValueAtTime(0.0001, t + dur + rel);

    const stopT = t + dur + rel + 0.05;
    sub.start(t);
    bow.start(t);
    bow2.start(t);
    sub.stop(stopT);
    bow.stop(stopT);
    bow2.stop(stopT);
  };

  // Instrument E: Horological Astrolabe Percussion (Escapement Tick, Frame Pulse, Celestial Chime)
  const playClockPerc = (t, kind, vel = 0.4, pan = 0, midi = 86) => {
    const panner = createPanner(pan);
    panner.connect(mixBus);

    if (kind === 'tick') {
      const src = ctx.createBufferSource();
      src.buffer = noiseBuf;
      const bp = ctx.createBiquadFilter();
      bp.type = 'bandpass';
      bp.frequency.value = 3400 + (pan > 0 ? 500 : -350);
      bp.Q.value = 4.5;
      const g = ctx.createGain();
      g.gain.setValueAtTime(0.06 * vel, t);
      g.gain.exponentialRampToValueAtTime(0.0001, t + 0.024);
      src.connect(bp);
      bp.connect(g);
      g.connect(panner);
      src.start(t);
      src.stop(t + 0.03);
    } else if (kind === 'pulse') {
      // Warm muffled orchestral timpani / frame-drum heartbeat
      const o = ctx.createOscillator();
      o.type = 'sine';
      o.frequency.setValueAtTime(118, t);
      o.frequency.exponentialRampToValueAtTime(52, t + 0.13);
      const g = ctx.createGain();
      g.gain.setValueAtTime(0.0001, t);
      g.gain.linearRampToValueAtTime(0.24 * vel, t + 0.005);
      g.gain.exponentialRampToValueAtTime(0.0001, t + 0.24);
      o.connect(g);
      g.connect(panner);

      const revTap = ctx.createGain();
      revTap.gain.value = 0.25;
      g.connect(revTap);
      revTap.connect(reverbSend);

      o.start(t);
      o.stop(t + 0.27);
    } else if (kind === 'chime') {
      // Shimmering astrolabe brass bell harmonic
      const f = midiToFreq(midi);
      const ratios = [1, 2.756, 5.404];
      const amps = [0.55, 0.26, 0.1];
      const decs = [2.2, 1.1, 0.5];
      const revTap = ctx.createGain();
      revTap.gain.value = 0.72;
      panner.connect(revTap);
      revTap.connect(reverbSend);

      ratios.forEach((r, idx) => {
        const o = ctx.createOscillator();
        o.type = 'sine';
        o.frequency.setValueAtTime(f * r, t);
        const g = ctx.createGain();
        const peak = 0.095 * vel * amps[idx];
        g.gain.setValueAtTime(0.0001, t);
        g.gain.linearRampToValueAtTime(peak, t + 0.004);
        g.gain.exponentialRampToValueAtTime(0.0001, t + decs[idx]);
        o.connect(g);
        g.connect(panner);
        o.start(t);
        o.stop(t + decs[idx] + 0.05);
      });
    }
  };

  // 3. THE COMPOSITION: 8 BARS (32 BEATS @ 128 BPM = 15.000s)
  //    D Minor Lament -> Contrapuntal Ascension -> Cathedral Climax -> Picardy D Major 9

  const events = []; // Stored for frame-locked visual choreography
  const beatToSec = (b) => b * BEAT;
  const beatToAudio = (b) => startTime + b * BEAT;

  const logEvent = (beat, voice, midi, vel, durBeats, extra = null) => {
    events.push({
      time: beatToSec(beat),
      beat,
      voice,
      midi,
      pc: ((midi % 12) + 12) % 12,
      vel,
      dur: beatToSec(durBeats),
      extra,
      triggered: false,
    });
  };

  // --- Harmonic Pads & Organ Pedal Bass ---
  const harmony = [
    // Act I: Awakening (Baroque Descending Lament Bass: D -> C -> Bb -> A)
    { b: 0,  dur: 2, bass: [38],     pad: [50, 57, 62, 64, 65], vel: 0.42, bVel: 0.50, bright: 0.78, label: 'Dm9' },

    // Act II: Cantus Ascension

      g.strokeRect(bx, by, boxW, boxH);

      // Center hairline axis
      g.strokeStyle = 'rgba(226, 194, 134, 0.12)';
      g.beginPath();
      g.moveTo(bx + 18, midY);
      g.lineTo(bx + boxW - 18, midY);
      g.stroke();

      // Corner brass diamonds on the cartouche
      const cPts = [[bx, by], [bx + boxW, by], [bx, by + boxH], [bx + boxW, by + boxH]];
      g.fillStyle = 'rgba(242, 212, 152, 0.65)';
      for (const [px, py] of cPts) {
        g.beginPath();
        g.moveTo(px, py - 4);
        g.lineTo(px + 4, py);
        g.lineTo(px, py + 4);
        g.lineTo(px - 4, py);
        g.closePath();
        g.fill();
      }

      // Live dual-voice luminous ribbon waveform
      g.globalCompositeOperation = 'lighter';
      const pts = 180;
      const waveHue = picardyMix > 0.35 ? 45 : (climaxMix > 0.35 ? 32 : 212);
      for (let pass = 0; pass < 3; pass++) {
        g.beginPath();
        for (let i = 0; i <= pts; i++) {
          const frac = i / pts;
          const x = bx + 20 + frac * (boxW - 40);
          const env = Math.sin(frac * Math.PI); // Smooth tapered ends
          const idx = Math.floor(frac * (waveData.length - 1));
          const sample = (waveData[idx] - 128) / 128;
          const counterSine = Math.sin(frac * Math.PI * 6 + t * 3.2) * (0.14 + smoothMid * 0.25);
          const amp = (pass === 2 ? -0.6 : 1.0) * (sample * 56 + counterSine * 18) * env;
          const y = midY + amp;
          if (i === 0) g.moveTo(x, y);
          else g.lineTo(x, y);
        }
        if (pass === 0) {
          g.strokeStyle = `hsla(${waveHue}, 88%, 68%, 0.22)`;
          g.lineWidth = 8.0;
        } else if (pass === 1) {
          g.strokeStyle = `hsla(${waveHue + 12}, 96%, 86%, 0.85)`;
          g.lineWidth = 1.9;
        } else {
          g.strokeStyle = `hsla(42, 90%, 76%, 0.42)`;
          g.lineWidth = 1.25;
        }
        g.stroke();
      }
      g.restore();

      // Temporal Progress Filament beneath cartouche
      const py = 1254;
      const px0 = bx;
      const px1 = bx + boxW;
      g.save();
      g.strokeStyle = 'rgba(222, 190, 128, 0.2)';
      g.lineWidth = 1.5;
      g.beginPath();
      g.moveTo(px0, py);
      g.lineTo(px1, py);
      g.stroke();

      const curX = px0 + boxW * progress;
      const pg = g.createLinearGradient(px0, py, curX, py);
      pg.addColorStop(0, 'rgba(182, 198, 240, 0.55)');
      pg.addColorStop(0.7, 'rgba(246, 206, 128, 0.88)');
      pg.addColorStop(1, 'rgba(255, 244, 204, 1.0)');
      g.strokeStyle = pg;
      g.lineWidth = 2.4;
      g.beginPath();
      g.moveTo(px0, py);
      g.lineTo(curX, py);
      g.stroke();

      // 4 Act landmark ticks along progress bar
      for (let i = 0; i <= 4; i++) {
        const mx = px0 + (boxW * i) / 4;
        const lit = progress >= i / 4;
        g.fillStyle = lit ? 'rgba(255, 232, 170, 0.92)' : 'rgba(218, 190, 134, 0.28)';
        g.beginPath();
        g.arc(mx, py, i === 3 ? 3.5 : 2.3, 0, Math.PI * 2);
        g.fill();
      }

      // Glowing progress cursor diamond
      g.fillStyle = '#fff5d6';
      g.beginPath();
      g.moveTo(curX, py - 5);
      g.lineTo(curX + 5, py);
      g.lineTo(curX, py + 5);
      g.lineTo(curX - 5, py);
      g.closePath();
      g.fill();

      // Museum-grade engraved title inscription
      g.font = '400 15px Georgia, "Times New Roman", serif';
      g.textAlign = 'center';
      g.textBaseline = 'middle';
      g.fillStyle = `rgba(232, 208, 162, ${0.42 + picardyMix * 0.26})`;
      g.fillText('N O C T U R N E   O F   T H E   B R A S S   A S T R O L A B E', CX, 1292);
      g.restore();
    }

    // --- LAYER 9: Picardy Third Solar Wash & Subtle Lithograph Grain ---
    if (picardyFlare > 0.01) {
      g.save();
      g.globalCompositeOperation = 'lighter';
      const sg = g.createRadialGradient(CX, CY, 10, CX, CY, 740);
      sg.addColorStop(0, `rgba(255, 244, 206, ${picardyFlare * 0.42})`);
      sg.addColorStop(0.5, `rgba(252, 204, 114, ${picardyFlare * 0.2})`);
      sg.addColorStop(1, 'rgba(252, 204, 114, 0)');
      g.fillStyle = sg;
      g.fillRect(0, 0, W, H);
      g.restore();
    }

    if (grainPattern) {
      g.save();
      g.fillStyle = grainPattern;
      g.fillRect(0, 0, W, H);
      g.restore();
    }

    // Gentle fade-in from twilight at start (0.0 -> 0.45s)
    if (t < 0.45) {
      g.fillStyle = `rgba(4, 6, 13, ${1 - smoothstep(0, 0.45, t)})`;
      g.fillRect(0, 0, W, H);
    }
  };

  // Frame-accurate audio-locked animation clock
  const loop = () => {
    const audioElapsed = ctx.currentTime - startTime;
    const perfElapsed = (performance.now() - perfStart) / 1000 - 0.045;
    const t = Math.max(0, audioElapsed > -0.02 ? audioElapsed : perfElapsed);
    drawFrame(Math.min(TOTAL_DUR, t));
    if (t < TOTAL_DUR + 0.25) {
      rafId = requestAnimationFrame(loop);
    }
  };

  // Render initial frame immediately and begin animation loop
  drawFrame(0);
  rafId = requestAnimationFrame(loop);

  return () => {
    if (rafId) cancelAnimationFrame(rafId);
  };
}
