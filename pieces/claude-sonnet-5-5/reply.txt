export const title = 'Descent in Three Tempi';

export function start({ audioContext, canvas, duration, random }) {
  const ctx = audioContext;
  const D = duration > 0 ? duration : 15;
  const W = canvas.width, H = canvas.height;
  const g = canvas.getContext('2d');
  const STEP = D / 128;                       // 128 sixteenths = 8 bars
  const TAU = Math.PI * 2;
  const mtof = (m) => 440 * Math.pow(2, (m - 69) / 12);
  const clamp = (x, a = 0, b = 1) => (x < a ? a : x > b ? b : x);
  const lerp = (a, b, t) => a + (b - a) * t;
  const smooth = (a, b, x) => { const u = clamp((x - a) / (b - a)); return u * u * (3 - 2 * u); };
  const easeOut = (u) => 1 - Math.pow(1 - clamp(u), 3);
  const rnd = (a = 0, b = 1) => a + (b - a) * random();
  try { if (ctx.state === 'suspended') ctx.resume(); } catch (e) {}
  const t0 = ctx.currentTime + 0.08;

  /* ───────────────────────── AUDIO GRAPH ───────────────────────── */
  const master = ctx.createGain(); master.gain.value = 0.85;
  const comp = ctx.createDynamicsCompressor();
  comp.threshold.value = -16; comp.knee.value = 18; comp.ratio.value = 3.5;
  comp.attack.value = 0.006; comp.release.value = 0.22;
  const fade = ctx.createGain();
  master.connect(comp); comp.connect(fade); fade.connect(ctx.destination);
  fade.gain.setValueAtTime(1, t0);
  fade.gain.setValueAtTime(1, t0 + D * 0.935);
  fade.gain.linearRampToValueAtTime(0, t0 + D - 0.02);

  // reverb (seeded noise impulse)
  const sr = ctx.sampleRate;
  const irLen = Math.floor(sr * 2.8);
  const ir = ctx.createBuffer(2, irLen, sr);
  for (let c = 0; c < 2; c++) {
    const d = ir.getChannelData(c); let lp = 0;
    for (let i = 0; i < irLen; i++) {
      const x = i / irLen; const n = random() * 2 - 1;
      lp += (n - lp) * (0.65 - 0.55 * x);
      d[i] = (i < sr * 0.012 ? 0 : 1) * lp * Math.pow(1 - x, 2.6) * 1.6;
    }
  }
  const rev = ctx.createConvolver(); rev.buffer = ir;
  const revOut = ctx.createGain(); revOut.gain.value = 0.9;
  rev.connect(revOut); revOut.connect(master);

  // ping-pong delay
  const hasPan = typeof ctx.createStereoPanner === 'function';
  const delIn = ctx.createGain();
  const dL = ctx.createDelay(1), dR = ctx.createDelay(1);
  dL.delayTime.value = STEP * 3; dR.delayTime.value = STEP * 2;
  const fb1 = ctx.createGain(), fb2 = ctx.createGain(); fb1.gain.value = 0.42; fb2.gain.value = 0.42;
  const dlp = ctx.createBiquadFilter(); dlp.type = 'lowpass'; dlp.frequency.value = 3200;
  delIn.connect(dL); dL.connect(fb1); fb1.connect(dR); dR.connect(fb2); fb2.connect(dlp); dlp.connect(dL);
  const delOut = ctx.createGain(); delOut.gain.value = 0.8; delOut.connect(master);
  if (hasPan) {
    const pl = ctx.createStereoPanner(), pr = ctx.createStereoPanner(); pl.pan.value = -0.7; pr.pan.value = 0.7;
    dL.connect(pl); dR.connect(pr); pl.connect(delOut); pr.connect(delOut);
  } else { dL.connect(delOut); dR.connect(delOut); }

  function makeBus(dry, rv, dl) {
    const inp = ctx.createGain();
    const d = ctx.createGain(); d.gain.value = dry; inp.connect(d); d.connect(master);
    if (rv) { const r = ctx.createGain(); r.gain.value = rv; inp.connect(r); r.connect(rev); }
    if (dl) { const e = ctx.createGain(); e.gain.value = dl; inp.connect(e); e.connect(delIn); }
    return inp;
  }
  const busLead = makeBus(0.9, 0.22, 0.05);
  const busTenor = makeBus(0.85, 0.3, 0);
  const busBell = makeBus(0.7, 0.4, 0.3);
  const busBass = makeBus(1, 0.04, 0);
  const busPad = makeBus(0.7, 0.5, 0);
  const busFx = makeBus(0.6, 0.5, 0.15);

  function route(node, bus, pan) {
    if (hasPan && pan) { const p = ctx.createStereoPanner(); p.pan.value = clamp(pan, -1, 1); node.connect(p); p.connect(bus); }
    else node.connect(bus);
  }
  const nbuf = ctx.createBuffer(1, sr * 2, sr);
  { const d = nbuf.getChannelData(0); for (let i = 0; i < d.length; i++) d[i] = random() * 2 - 1; }

  function pluck(t, midi, dur, vel, pan) {            // harpsichord-ish
    const f = mtof(midi);
    const o1 = ctx.createOscillator(), o2 = ctx.createOscillator();
    o1.type = 'sawtooth'; o2.type = 'square';
    o1.frequency.value = f; o2.frequency.value = f; o2.detune.value = 7;
    const m2 = ctx.createGain(); m2.gain.value = 0.35;
    const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.Q.value = 0.8;
    lp.frequency.setValueAtTime(Math.min(f * 9, 11000), t);
    lp.frequency.exponentialRampToValueAtTime(Math.min(Math.max(f * 2.5, 600), 8000), t + 0.3);

    const pk = 0.2 * vel, hold = Math.max(dur, 0.12);
    env.gain.setValueAtTime(0.0001, t);
    env.gain.linearRampToValueAtTime(pk, t + 0.003);
    env.gain.exponentialRampToValueAtTime(pk * 0.25, t + 0.003 + hold);
    env.gain.exponentialRampToValueAtTime(0.0001, t + 0.003 + hold + 0.18);
    o1.connect(lp); o2.connect(m2); m2.connect(lp); lp.connect(env);
    route(env, busLead, pan);
    const end = t + hold + 0.3;
    o1.start(t); o2.start(t); o1.stop(end); o2.stop(end);
    const n = ctx.createBufferSource(); n.buffer = nbuf;
    const hp = ctx.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 3000;
    const ng = ctx.createGain();
    ng.gain.setValueAtTime(0.05 * vel, t); ng.gain.exponentialRampToValueAtTime(0.0001, t + 0.02);
    n.connect(hp); hp.connect(ng); route(ng, busLead, pan);
    n.start(t, rnd(0, 1.5)); n.stop(t + 0.03);
  }
  function bow(t, midi, dur, vel, pan) {              // viola-ish tenor

    const env = ctx.createGain();

    env.gain.setValueAtTime(0.0001, t);
    env.gain.linearRampToValueAtTime(pk, t + 0.06);
    env.gain.setValueAtTime(pk, end);
    env.gain.linearRampToValueAtTime(0.0001, end + rel);
    lp.connect(env); route(env, busTenor, pan);

    const lg = ctx.createGain(); lg.gain.setValueAtTime(0, t); lg.gain.linearRampToValueAtTime(9, t + 0.5);
    lfo.connect(lg);
    const stopT = end + rel + 0.05;
    for (const det of [-7, 7]) {
      const o = ctx.createOscillator(); o.type = 'sawtooth'; o.frequency.value = f; o.detune.value = det;
      lg.connect(o.detune); o.connect(lp); o.start(t); o.stop(stopT);
    }
    const o3 = ctx.createOscillator(); o3.type = 'triangle'; o3.frequency.value = f;
    const g3 = ctx.createGain(); g3.gain.value = 0.8; o3.connect(g3); g3.connect(lp); o3.start(t); o3.stop(stopT);
    lfo.start(t); lfo.stop(stopT);
  }
  function bell(t, midi, vel, dec, pan) {             // glassy celesta
    const f = mtof(midi);
    const env = ctx.createGain(), pk = 0.17 * vel;
    env.gain.setValueAtTime(0.0001, t);
    env.gain.linearRampToValueAtTime(pk, t + 0.002);
    env.gain.exponentialRampToValueAtTime(0.0001, t + dec);

    const o3 = ctx.createOscillator(); o3.type = 'sine'; o3.frequency.value = f * 5.43;
    const g2 = ctx.createGain(); g2.gain.value = 0.22;
    const g3 = ctx.createGain(); g3.gain.value = 0.1;
    const e3 = ctx.createGain(); e3.gain.setValueAtTime(1, t); e3.gain.exponentialRampToValueAtTime(0.001, t + 0.12);
    o1.connect(env); o2.connect(g2); g2.connect(env); o3.connect(g3); g3.connect(e3); e3.connect(env);
    route(env, busBell, pan);
    const end = t + dec + 0.05;
    o1.start(t); o2.start(t); o3.start(t); o1.stop(end); o2.stop(end); o3.stop(end);
  }
  function bassNote(t, midi, dur, vel) {
    const f = mtof(midi);
    const o = ctx.createOscillator(); o.type = 'sawtooth'; o.frequency.value = f;
    const s = ctx.createOscillator(); s.type = 'sine'; s.frequency.value = f;
    const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 380; lp.Q.value = 0.7;
    const gs = ctx.createGain(); gs.gain.value = 0.9;
    const env = ctx.createGain(), pk = 0.26 * vel;
    env.gain.setValueAtTime(0.0001, t);
    env.gain.linearRampToValueAtTime(pk, t + 0.03);
    env.gain.setValueAtTime(pk, t + dur - 0.05);
    env.gain.linearRampToValueAtTime(0.0001, t + dur + 0.25);
    o.connect(lp); lp.connect(env); s.connect(gs); gs.connect(env);
    route(env, busBass, 0);
    o.start(t); s.start(t); o.stop(t + dur + 0.3); s.stop(t + dur + 0.3);
  }
  function bassPluck(t, midi, vel) {

    const o = ctx.createOscillator(); o.type = 'triangle'; o.frequency.value = f;
    const o2 = ctx.createOscillator(); o2.type = 'sawtooth'; o2.frequency.value = f;
    const m2 = ctx.createGain(); m2.gain.value = 0.3;
    const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 900;
    const env = ctx.createGain(), pk = 0.17 * vel;
    env.gain.setValueAtTime(0.0001, t);
    env.gain.linearRampToValueAtTime(pk, t + 0.005);
    env.gain.exponentialRampToValueAtTime(0.0001, t + 0.2);
    o.connect(lp); o2.connect(m2); m2.connect(lp); lp.connect(env); route(env, busBass, 0);
    o.start(t); o2.start(t); o.stop(t + 0.25); o2.stop(t + 0.25);
  }
  function padChord(t, midis, dur, vol) {
    const lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.Q.value = 0.6;
    lp.frequency.setValueAtTime(500, t); lp.frequency.linearRampToValueAtTime(1500, t + dur);
    const env = ctx.createGain();
    env.gain.setValueAtTime(0.0001, t);
    env.gain.linearRampToValueAtTime(vol, t + 0.45);
    env.gain.setValueAtTime(vol, t + dur - 0.1);
    env.gain.linearRampToValueAtTime(0.0001, t + dur + 0.5);
    lp.connect(env); route(env, busPad, 0);
    for (const m of midis) for (const det of [-6, 6]) {
      const o = ctx.createOscillator(); o.type = 'sawtooth'; o.frequency.value = mtof(m); o.detune.value = det;
      o.connect(lp); o.start(t); o.stop(t + dur + 0.6);
    }
  }
  function boom(t) {
    const o = ctx.createOscillator(); o.type = 'sine';
    o.frequency.setValueAtTime(120, t); o.frequency.exponentialRampToValueAtTime(36.7, t + 0.3);
    const env = ctx.createGain();
    env.gain.setValueAtTime(0.0001, t); env.gain.linearRampToValueAtTime(0.85, t + 0.006);
    env.gain.exponentialRampToValueAtTime(0.0001, t + 2.2);
    o.connect(env); route(env, busBass, 0); o.start(t); o.stop(t + 2.3);
  }
  function riser(tS, tE) {
    const ns = ctx.createBufferSource(); ns.buffer = nbuf; ns.loop = true;
    const bp = ctx.createBiquadFilter(); bp.type = 'bandpass'; bp.Q.value = 1.5;
    bp.frequency.setValueAtTime(400, tS); bp.frequency.exponentialRampToValueAtTime(7000, tE);
    bp.frequency.exponentialRampToValueAtTime(9500, tE + 2);
    const ng = ctx.createGain();
    ng.gain.setValueAtTime(0.0001, tS); ng.gain.exponentialRampToValueAtTime(0.13, tE);
    ng.gain.exponentialRampToValueAtTime(0.0001, tE + 2.0);
    ns.connect(bp); bp.connect(ng); route(ng, busFx, 0);
    ns.start(tS, 0); ns.stop(tE + 2.1);
  }

  /* ───────────────────────── COMPOSITION ─────────────────────────
     A chromatic lament bass (D–C♯–C–B–B♭–A) in 8 bars.
     One motif is sounded in three simultaneous tempi:

       tenor = augmentation, ×2     (bowed)
       bells = inversion, diminution ÷2 (mirror image, twice as fast)
     Every beat is snapped to the current chord, so the counterpoint
     stays consonant while the sequence descends with the bass.
     Bar 7 accelerates to a 32nd-note run, bar 8 resolves to a

    A7:  { tones: [9, 1, 4, 7],   scale: [9, 11, 1, 2, 4, 6, 7] },
    Dm7: { tones: [0, 2, 5, 9],   scale: [0, 2, 4, 5, 7, 9, 10] },
    Bo:  { tones: [11, 2, 5, 9],  scale: [11, 0, 2, 4, 5, 7, 9] },
    Gm7: { tones: [7, 10, 2, 5],  scale: [7, 9, 10, 0, 2, 4, 5] },
    F6:  { tones: [9, 0, 5, 2],   scale: [5, 7, 9, 10, 0, 2, 4] },

  const H8 = M.filter((n) => n[0] < 8);

  const EV = [[], [], [], [], []];
  function addNote(v, step, midi, durSteps, vel) {
    const t = t0 + step * STEP + random() * 0.004;
    const dur = durSteps * STEP;
    const pan = v === 0 ? -0.22 : v === 1 ? 0.18 : (random() * 2 - 1) * 0.6;
    if (v === 0) pluck(t, midi, dur, vel, pan);
    else if (v === 1) bow(t, midi, dur, vel, pan);
    else if (v === 2) bell(t, midi, vel, 0.9, pan);
    else if (v === 4) bell(t, midi, vel, 2.2, pan);
    EV[v].push({ t: step * STEP, dur, midi, vel, step });
  }
  function snap(target, pcs, prev, lo, hi) {
    let best = null, bc = 1e9;
    for (let m = Math.floor(target) - 9; m <= Math.ceil(target) + 9; m++) {
      if (m < lo || m > hi) continue;
      if (pcs.indexOf(((m % 12) + 12) % 12) < 0) continue;
      const c = Math.abs(m - target) - (m > target ? 0.02 : 0) + (m === prev ? 0.75 : 0);
      if (c < bc) { bc = c; best = m; }
    }
    return best === null ? Math.round(target) : best;
  }
  const prevV = [-1, -1, -1];
  function statement(v, start, notes, scale, offset, sign, lo, hi) {
    for (const [st, du, semi] of notes) {
      const step = start + st * scale;
      if (step >= 128) continue;
      const sl = Math.min(15, Math.floor(step / 8));
      const ch = slotChord[sl];
      const target = slotBass[sl] + offset + sign * semi;
      const strong = step % 4 === 0;
      const m = snap(target, strong ? ch.tones : ch.scale, prevV[v], lo, hi);
      prevV[v] = m;
      const r = step / 128;
      const vel = v === 0 ? (0.5 + 0.42 * r) * (strong ? 1 : 0.8)
        : v === 1 ? 0.72 + 0.2 * r
        : (0.4 + 0.38 * r) * (strong ? 1 : 0.85);
      addNote(v, step, m, du * scale, vel);
    }
  }

  const KEYS = [[0, 440], [16, 520], [32, 600], [48, 680], [64, 760], [80, 840], [96, 1000], [104, 840], [112, 640]];
  function sunY(t) {
    let k = -1;
    for (let i = 0; i < KEYS.length; i++) if (t >= KEYS[i][0] * STEP) k = i;
    if (k < 0) return 440;
    const y1 = KEYS[k][1], y0 = k > 0 ? KEYS[k - 1][1] : y1;
    const a = t - KEYS[k][0] * STEP;
    const d = (y0 - y1) * Math.exp(-6 * a) * (Math.cos(10 * a) + 0.6 * Math.sin(10 * a));
    return y1 + d + 7 * Math.sin(t * 1.7);
  }
  const TK = [[0, [16, 44, 104]], [4, [34, 40, 124]], [8, [104, 30, 112]], [11.5, [170, 36, 76]],
    [13.125, [255, 150, 64]], [15, [255, 196, 112]]];

  const TB = 112 * STEP;

    }

      g.fillRect(s.x, y, s.s, s.s);

        for (const m of e.emb) {
          if (a > m.life) continue;

          g.strokeStyle = rgba(R.col, (1 - q.age / 0.45) * 0.5 * q.vel * ra);

    g.restore();

    // title
    try { g.letterSpacing = '10px'; } catch (e) {}
    g.textAlign = 'center'; g.textBaseline = 'alphabetic';
    g.font = '600 24px "Helvetica Neue", Helvetica, Arial, sans-serif';
    g.fillStyle = 'rgba(255,236,205,0.5)';
    g.fillText('DESCENT IN THREE TEMPI', W / 2, 1290);
    try { g.letterSpacing = '4px'; } catch (e) {}
    g.font = '14px "Helvetica Neue", Helvetica, Arial, sans-serif';
    g.fillStyle = 'rgba(255,236,205,0.3)';

  }

  /* ───────────────────────── LOOP ───────────────────────── */
  let stopped = false, rafId = 0;

  const caf = (typeof cancelAnimationFrame === 'function') ? cancelAnimationFrame : clearTimeout;

    if (stopped) return;
    const t = clamp(ctx.currentTime - t0, 0, D);
    draw(t);
    rafId = raf(frame);
  }
  frame();

  return function stop() {
    stopped = true;
    try { caf(rafId); } catch (e) {}
    try { fade.disconnect(); } catch (e) {}
  };
}

**Descent in Three Tempi** is a 15-second piece in D minor that resolves to a Picardy D major. I haven't run it, so the sound and picture are untested.

**Music:** A chromatic lament bass (D–C♯–C–B–B♭–A–G–A–D) runs across 8 bars, and one 1-bar motif is heard in three tempi at once:
- A harpsichord-like lead plays the motif at normal speed, sequencing down with the bass.
- A bowed tenor plays it in augmentation (half speed), entering in bar 3.
- Glassy bells play it inverted and in diminution (double speed), entering in bar 5.

The motif's beats are snapped to the current chord, so the counterpoint stays consonant. Bar 7 builds with a noise riser and a rising 32nd-note run. Bar 8 lands on D major with a low boom and a shower of bell sparkles.

**Picture:** A sun sinks stepwise with the bass, bouncing on a spring at each bar change. Three rings turn around it at 2:1:½ speeds, and each ring's necklace of beads traces its voice's motif. The bell ring turns the opposite way, as the mirror of the lead. Each note sheds embers that fall, there is a bass-note ladder on the right edge, and the colour moves from cold blue through crimson to gold. At the Picardy arrival the sun rises into a burst of rays.
